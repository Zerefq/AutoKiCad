// ---- js/util.js ----
// Общие вспомогательные функции
window.AK = window.AK || {};

AK.util = (function () {
  let counter = 0;

  function uid(prefix) {
    counter++;
    return prefix + Date.now().toString(36) + counter.toString(36) + Math.random().toString(36).slice(2, 5);
  }

  function uuid() {
    if (window.crypto && crypto.randomUUID) {
      try { return crypto.randomUUID(); } catch (e) { /* небезопасный контекст */ }
    }
    return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, c => {
      const r = Math.random() * 16 | 0;
      return (c === 'x' ? r : (r & 0x3 | 0x8)).toString(16);
    });
  }

  const MULT = { p: 1e-12, n: 1e-9, u: 1e-6, 'µ': 1e-6, 'μ': 1e-6, m: 1e-3, k: 1e3, K: 1e3, M: 1e6, G: 1e9, R: 1, r: 1 };

  // "4.7k" -> 4700, "4k7" -> 4700, "100n" -> 1e-7, "220" -> 220
  function parseValue(s) {
    if (s == null) return NaN;
    s = String(s).trim().replace(/\s+/g, '').replace(/(Ω|ohm|Ом|F|Ф|H|Гн|V|В)$/i, '');
    const m = s.match(/^(\d*\.?\d+)([pnuµμmkKMGRr]?)(\d*)$/);
    if (!m) return NaN;
    let num = m[3] ? parseFloat(m[1] + '.' + m[3]) : parseFloat(m[1]);
    if (m[2]) num *= MULT[m[2]];
    return num;
  }

  function fmt(n, digits) {
    const p = Math.pow(10, digits == null ? 4 : digits);
    const v = Math.round(n * p) / p;
    return (Object.is(v, -0) ? 0 : v).toString();
  }

  function download(name, text, mime) {
    const blob = text instanceof Blob ? text : new Blob([text], { type: mime || 'text/plain;charset=utf-8' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = name;
    document.body.appendChild(a);
    a.click();
    setTimeout(() => { URL.revokeObjectURL(a.href); a.remove(); }, 500);
  }

  function esc(s) {
    return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
  }

  function rotPt(x, y, deg) {
    const r = ((deg % 360) + 360) % 360;
    if (r === 0) return [x, y];
    if (r === 90) return [-y, x];
    if (r === 180) return [-x, -y];
    if (r === 270) return [y, -x];
    const a = r * Math.PI / 180;
    return [x * Math.cos(a) - y * Math.sin(a), x * Math.sin(a) + y * Math.cos(a)];
  }

  // Минимальный ZIP без сжатия (store). files: [[путь, текст], ...] -> Uint8Array
  let crcTable = null;
  function crc32(bytes) {
    if (!crcTable) {
      crcTable = new Uint32Array(256);
      for (let n = 0; n < 256; n++) {
        let c = n;
        for (let k = 0; k < 8; k++) c = c & 1 ? 0xEDB88320 ^ (c >>> 1) : c >>> 1;
        crcTable[n] = c >>> 0;
      }
    }
    let c = 0xFFFFFFFF;
    for (let i = 0; i < bytes.length; i++) c = crcTable[(c ^ bytes[i]) & 0xFF] ^ (c >>> 8);
    return (c ^ 0xFFFFFFFF) >>> 0;
  }

  function zip(files) {
    const enc = new TextEncoder();
    const parts = [], central = [];
    let offset = 0;
    const d = new Date();
    const dosTime = (d.getHours() << 11) | (d.getMinutes() << 5) | (d.getSeconds() >> 1);
    const dosDate = ((d.getFullYear() - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate();
    files.forEach(([name, text]) => {
      const nameB = enc.encode(name), data = enc.encode(text), crc = crc32(data);
      const local = new DataView(new ArrayBuffer(30));
      local.setUint32(0, 0x04034b50, true); local.setUint16(4, 20, true); local.setUint16(6, 0x0800, true);
      local.setUint16(8, 0, true); local.setUint16(10, dosTime, true); local.setUint16(12, dosDate, true);
      local.setUint32(14, crc, true); local.setUint32(18, data.length, true); local.setUint32(22, data.length, true);
      local.setUint16(26, nameB.length, true); local.setUint16(28, 0, true);
      const cen = new DataView(new ArrayBuffer(46));
      cen.setUint32(0, 0x02014b50, true); cen.setUint16(4, 20, true); cen.setUint16(6, 20, true); cen.setUint16(8, 0x0800, true);
      cen.setUint16(10, 0, true); cen.setUint16(12, dosTime, true); cen.setUint16(14, dosDate, true);
      cen.setUint32(16, crc, true); cen.setUint32(20, data.length, true); cen.setUint32(24, data.length, true);
      cen.setUint16(28, nameB.length, true); cen.setUint32(42, offset, true);
      parts.push(new Uint8Array(local.buffer), nameB, data);
      central.push(new Uint8Array(cen.buffer), nameB);
      offset += 30 + nameB.length + data.length;
    });
    const cenSize = central.reduce((a, b) => a + b.length, 0);
    const end = new DataView(new ArrayBuffer(22));
    end.setUint32(0, 0x06054b50, true); end.setUint16(8, files.length, true); end.setUint16(10, files.length, true);
    end.setUint32(12, cenSize, true); end.setUint32(16, offset, true);
    const all = [...parts, ...central, new Uint8Array(end.buffer)];
    const out = new Uint8Array(all.reduce((a, b) => a + b.length, 0));
    let p = 0;
    all.forEach(b => { out.set(b, p); p += b.length; });
    return out;
  }

  // SVG-текст → PNG (Blob); scale — во сколько раз крупнее размеров SVG
  function svgToPng(svgText, scale) {
    return new Promise((resolve, reject) => {
      const url = URL.createObjectURL(new Blob([svgText], { type: 'image/svg+xml' }));
      const img = new Image();
      img.onload = () => {
        const cv = document.createElement('canvas');
        cv.width = Math.round(img.naturalWidth * (scale || 1));
        cv.height = Math.round(img.naturalHeight * (scale || 1));
        const g = cv.getContext('2d');
        g.fillStyle = '#fff';
        g.fillRect(0, 0, cv.width, cv.height);
        g.drawImage(img, 0, 0, cv.width, cv.height);
        URL.revokeObjectURL(url);
        cv.toBlob(b => (b ? resolve(b) : reject(new Error('не удалось создать картинку'))), 'image/png');
      };
      img.onerror = () => { URL.revokeObjectURL(url); reject(new Error('SVG не загрузился')); };
      img.src = url;
    });
  }

  return { uid, uuid, parseValue, fmt, download, esc, rotPt, zip, svgToPng };
})();

window.AK = window.AK || {};

AK.symBBox = function (sym) {
  let x1 = Infinity, y1 = Infinity, x2 = -Infinity, y2 = -Infinity;
  const add = (x, y) => { x1 = Math.min(x1, x); y1 = Math.min(y1, y); x2 = Math.max(x2, x); y2 = Math.max(y2, y); };
  sym.gfx.forEach(g => {
    if (g.pts) g.pts.forEach(p => add(p[0], p[1]));
    if (g.t === 'rect') { add(g.x1, g.y1); add(g.x2, g.y2); }
    if (g.t === 'circle') { add(g.cx - g.r, g.cy - g.r); add(g.cx + g.r, g.cy + g.r); }
  });
  Object.values(sym.pins).forEach(p => add(p.x, p.y));
  return { x1, y1, x2, y2 };
};

    const ohms = AK.util.parseValue(value);
    if (!(ohms > 0)) return [BAND[0], BAND[0], BAND[0]];

    let sig = Math.round(ohms / Math.pow(10, exp));
    if (sig >= 100) { sig = 10; exp++; }
    const mult = exp >= 0 ? BAND[Math.min(exp, 9)] : (exp === -1 ? '#c9a227' : '#b8b8b8');
    return [BAND[Math.floor(sig / 10)], BAND[sig % 10], mult];

    const f = AK.util.parseValue(value);

AK.pinName = function (type, pin) {
  const d = AK.LIB[type];
  if (d.sym && d.sym.pins[pin] && d.sym.pins[pin].name) return d.sym.pins[pin].name;
  if (d.pinNames && d.pinNames[pin]) return d.pinNames[pin];

};

// Позиция вывода компонента в мировых координатах режима «Монтаж»
AK.pinWorld = function (comp, pinId) {
  const p = AK.LIB[comp.type].visual.pins[pinId];
  const [rx, ry] = AK.util.rotPt(p[0], p[1], comp.rot || 0);
  return [comp.x + rx, comp.y + ry];
};

// Привязка к сетке 0.1": выводы (а не центр) компонента попадают в отверстия
AK.snapPos = function (type, rot, x, y) {
  const p0 = Object.values(AK.LIB[type].visual.pins)[0];
  const [ox, oy] = AK.util.rotPt(p0[0], p0[1], rot || 0);
  return [Math.round((x + ox) / 10) * 10 - ox, Math.round((y + oy) / 10) * 10 - oy];
};

AK.compWorldBox = function (comp) {
  const [bx, by, bw, bh] = AK.LIB[comp.type].visual.box;
  const pts = [[bx, by], [bx + bw, by], [bx, by + bh], [bx + bw, by + bh]].map(([x, y]) => AK.util.rotPt(x, y, comp.rot || 0));
  const xs = pts.map(p => p[0]), ys = pts.map(p => p[1]);
  return { x: comp.x + Math.min(...xs), y: comp.y + Math.min(...ys), w: Math.max(...xs) - Math.min(...xs), h: Math.max(...ys) - Math.min(...ys) };
};

// ---- js/easyeda-lib.js ----
// Компоненты из библиотеки EasyEDA (JLCEDA/EasyEDA Official Library, https://lceda.cn/ · https://easyeda.com).
// Из данных компонента строятся: символ схемы (sym), вид на макетной плате (visual, по посадочному месту)
// и сведения для вставки «родного» символа EasyEDA (ee).
// Единицы EasyEDA: схема — 10 = 100 mil = 1 U; посадочное место — 10 = 100 mil = 10 px монтажа.
window.AK = window.AK || {};

AK.EE = (function () {
  const API = 'https://easyeda.com';
  const VER = '6.5.51';
  const LCSC_UID = '0819f05c4eef4c71ace90d822a990e87';
  const R3 = v => Math.round(v * 1000) / 1000;

  // ---------- SVG path → ломаные (M L H V A C Q Z) ----------
  function pathToPolylines(d) {
    const tok = String(d).match(/[a-zA-Z]|-?\d*\.?\d+(?:e-?\d+)?/g) || [];
    const out = [];
    let i = 0, cmd = '', cur = [0, 0], start = [0, 0], line = null;
    const num = () => +tok[i++];
    const push = p => { if (!line) { line = [cur.slice()]; out.push(line); } line.push(p); cur = p; };
    while (i < tok.length) {
      if (/[a-zA-Z]/.test(tok[i])) cmd = tok[i++];
      const rel = cmd === cmd.toLowerCase();
      const C = cmd.toUpperCase();
      const base = rel ? cur : [0, 0];
      if (C === 'M') { cur = [base[0] + num(), base[1] + num()]; start = cur; line = null; cmd = rel ? 'l' : 'L'; }
      else if (C === 'L') push([base[0] + num(), base[1] + num()]);
      else if (C === 'H') push([(rel ? cur[0] : 0) + num(), cur[1]]);
      else if (C === 'V') push([cur[0], (rel ? cur[1] : 0) + num()]);
      else if (C === 'Z') { if (line) push(start.slice()); line = null; }
      else if (C === 'C') { num(); num(); num(); num(); push([base[0] + num(), base[1] + num()]); }
      else if (C === 'Q' || C === 'S') { num(); num(); push([base[0] + num(), base[1] + num()]); }
      else if (C === 'A') {
        const rx = Math.abs(num()), ry = Math.abs(num()), rot = num(), large = num(), sweep = num();
        const p = [base[0] + num(), base[1] + num()];
        arcPoints(cur, p, rx, ry, rot, large, sweep).forEach(push);
      } else i++;
    }
    return out.filter(l => l.length > 1);
  }

  // дуга SVG по конечным точкам → точки (центральная параметризация)
  function arcPoints(p1, p2, rx, ry, phiDeg, fa, fs) {
    if (!rx || !ry) return [p2];
    const phi = phiDeg * Math.PI / 180, cs = Math.cos(phi), sn = Math.sin(phi);
    const dx = (p1[0] - p2[0]) / 2, dy = (p1[1] - p2[1]) / 2;
    const x1 = cs * dx + sn * dy, y1 = -sn * dx + cs * dy;
    let lam = x1 * x1 / (rx * rx) + y1 * y1 / (ry * ry);
    if (lam > 1) { rx *= Math.sqrt(lam); ry *= Math.sqrt(lam); }
    let co = Math.sqrt(Math.max(0, (rx * rx * ry * ry - rx * rx * y1 * y1 - ry * ry * x1 * x1) / (rx * rx * y1 * y1 + ry * ry * x1 * x1)));
    if (fa === fs) co = -co;
    const cxp = co * rx * y1 / ry, cyp = -co * ry * x1 / rx;
    const cx = cs * cxp - sn * cyp + (p1[0] + p2[0]) / 2, cy = sn * cxp + cs * cyp + (p1[1] + p2[1]) / 2;
    const ang = (ux, uy, vx, vy) => Math.atan2(ux * vy - uy * vx, ux * vx + uy * vy);
    const t1 = ang(1, 0, (x1 - cxp) / rx, (y1 - cyp) / ry);
    let dt = ang((x1 - cxp) / rx, (y1 - cyp) / ry, (-x1 - cxp) / rx, (-y1 - cyp) / ry);
    if (!fs && dt > 0) dt -= 2 * Math.PI;
    if (fs && dt < 0) dt += 2 * Math.PI;
    const n = Math.max(4, Math.ceil(Math.abs(dt) / (Math.PI / 12)));
    const pts = [];
    for (let k = 1; k <= n; k++) {
      const t = t1 + dt * k / n;
      pts.push([cx + rx * Math.cos(t) * cs - ry * Math.sin(t) * sn, cy + rx * Math.cos(t) * sn + ry * Math.sin(t) * cs]);
    }
    pts[pts.length - 1] = p2;
    return pts;
  }

  // ---------- символ ----------
  const ELEC = { 1: 'in', 2: 'out', 4: 'pwr' };

  function parsePins(shapes) {
    const pins = [];
    shapes.filter(s => s.startsWith('P~')).forEach(s => {
      const seg = s.split('^^');
      const f = seg[0].split('~');
      const x = +f[4], y = +f[5];
      const m = (seg[2] || '').split('~')[0].match(/M\s*(-?[\d.]+)[\s,]+(-?[\d.]+)\s*([hvHVLl])\s*(-?[\d.]+)(?:[\s,]+(-?[\d.]+))?/);
      let dx = 0, dy = 0;
      if (m) {
        const c = m[3];
        if (c === 'h' || c === 'H') dx = c === 'h' ? +m[4] : +m[4] - +m[1];
        else if (c === 'v' || c === 'V') dy = c === 'v' ? +m[4] : +m[4] - +m[2];
        else if (c === 'L') { dx = +m[4] - +m[1]; dy = +m[5] - +m[2]; }
        else { dx = +m[4]; dy = +m[5]; }
      }
      const name = (seg[3] || '').split('~');
      const nm = (seg[4] || '').split('~');
      pins.push({
        num: f[3], x, y, dx, dy, hidden: f[1] !== 'show',
        name: name[4] || '', showName: name[0] === '1', showNum: nm[0] === '1', type: ELEC[f[2]],
      });
    });
    return pins;
  }

  function buildSym(ds) {
    const head = ds.head;
    const pins = parsePins(ds.shape);
    // начало координат символа выбираем так, чтобы выводы попали на целую сетку U
    const fr = v => ((Math.round((v - head.x) * 10) / 100) % 1 + 1) % 1;
    const best = arr => { const cnt = new Map(); arr.forEach(v => { const k = Math.round(v * 10) / 10 % 1; cnt.set(k, (cnt.get(k) || 0) + 1); }); return [...cnt].sort((a, b) => b[1] - a[1])[0][0]; };
    const fx = pins.length ? best(pins.map(p => fr(p.x))) : 0;
    const fy = pins.length ? best(pins.map(p => ((Math.round((p.y - head.y) * 10) / 100) % 1 + 1) % 1)) : 0;
    const ox = head.x + fx * 10, oy = head.y + fy * 10;
    const U = (x, y) => [R3((x - ox) / 10), R3((y - oy) / 10)];

    const gfx = [];
    ds.shape.forEach(s => {
      const f = s.split('~');
      const t = f[0];
      if (t === 'R') {
        const [x1, y1] = U(+f[1], +f[2]), w = +f[5] / 10, h = +f[6] / 10;
        gfx.push({ t: 'rect', x1, y1, x2: R3(x1 + w), y2: R3(y1 + h), fill: f[10] && f[10] !== 'none' ? 'body' : 'none' });
      } else if (t === 'E' || t === 'C') {
        const [cx, cy] = U(+f[1], +f[2]);
        const rx = +f[3] / 10, ry = (t === 'E' ? +f[4] : +f[3]) / 10;
        if (Math.abs(rx - ry) < 0.01) gfx.push({ t: 'circle', cx, cy, r: R3(rx), fill: 'none' });
        else { const pts = []; for (let k = 0; k <= 24; k++) { const a = k / 24 * 2 * Math.PI; pts.push([R3(cx + rx * Math.cos(a)), R3(cy + ry * Math.sin(a))]); } gfx.push({ t: 'line', pts }); }
      } else if (t === 'PL' || t === 'PG') {
        const nums = f[1].trim().split(/[\s,]+/).map(Number);
        const pts = [];
        for (let k = 0; k + 1 < nums.length; k += 2) pts.push(U(nums[k], nums[k + 1]));
        if (pts.length > 1) gfx.push(t === 'PG' && f[5] && f[5] !== 'none' ? { t: 'poly', pts } : { t: 'line', pts: t === 'PG' ? pts.concat([pts[0]]) : pts });
      } else if (t === 'A' || t === 'PT') {
        pathToPolylines(f[1]).forEach(l => gfx.push({ t: 'line', pts: l.map(p => U(p[0], p[1])) }));
      }
    });

    const DIR = (dx, dy) => Math.abs(dx) >= Math.abs(dy) ? (dx >= 0 ? 'R' : 'L') : (dy >= 0 ? 'D' : 'U');
    const spins = {};
    let showNames = false, showNums = false;
    pins.forEach(p => {
      const [ux, uy] = U(p.x, p.y);
      const sp = { x: Math.round(ux), y: Math.round(uy), dir: DIR(p.dx, p.dy), len: R3(Math.max(0.3, Math.hypot(p.dx, p.dy) / 10)), name: p.name, ee: [p.x, p.y] };
      if (p.type) sp.type = p.type;
      // вывод не на сетке — подпорка проводом при вставке в EasyEDA
      if (Math.abs(sp.x - ux) > 0.01 || Math.abs(sp.y - uy) > 0.01) sp.off = true;
      if (p.showName && p.name && p.name !== p.num) showNames = true;
      if (p.showNum) showNums = true;
      if (!spins[p.num]) spins[p.num] = sp;
    });
    const sym = { gfx, pins: spins, showNames, showNums: showNums && pins.length > 2, ic: pins.length > 4 };
    const bb = AK.symBBox(sym);
    sym.ref = [R3((bb.x1 + bb.x2) / 2), R3(bb.y1 - 0.6), 'middle'];
    sym.val = [R3((bb.x1 + bb.x2) / 2), R3(bb.y2 + 0.9), 'middle'];
    sym.bbox = AK.symBBox(sym);
    return { sym, origin: [ox, oy] };
  }

  // ---------- вид на макетной плате ----------
  function footprintPads(pk) {
    if (!pk || !pk.shape) return [];
    return pk.shape.filter(s => s.startsWith('PAD~')).map(s => {
      const f = s.split('~');
      return { num: f[8], x: +f[2], y: +f[3], tht: +f[9] > 0 };
    });
  }

  // раскладка выводов: по посадочному месту, если оно выводное и с шагом 0.1", иначе — запасная
  function pinLayout(nums, pads) {
    const tht = pads.filter(p => p.tht && nums.includes(p.num));
    const uniq = new Map();
    tht.forEach(p => { if (!uniq.has(p.num)) uniq.set(p.num, p); });
    if (uniq.size === nums.length && nums.length) {
      const list = [...uniq.values()], p0 = list[0];
      const pos = list.map(p => [Math.round((p.x - p0.x) / 10) * 10, Math.round((p.y - p0.y) / 10) * 10, p]);
      const onGrid = list.every(p => Math.abs((p.x - p0.x) / 10 - Math.round((p.x - p0.x) / 10)) < 0.2 && Math.abs((p.y - p0.y) / 10 - Math.round((p.y - p0.y) / 10)) < 0.2);
      const distinct = new Set(pos.map(q => q[0] + ',' + q[1])).size === pos.length;
      if (onGrid && distinct) {
        const xs = pos.map(q => q[0]), ys = pos.map(q => q[1]);
        const cx = Math.round((Math.min(...xs) + Math.max(...xs)) / 20) * 10, cy = Math.round((Math.min(...ys) + Math.max(...ys)) / 20) * 10;
        const pins = {};
        pos.forEach(([x, y, p]) => { pins[p.num] = [x - cx, y - cy]; });
        return { pins, fromFootprint: true };
      }
    }
    // запасной вариант (SMD, шаг 1.27 мм и т.п.): выводы с шагом 0.1", как у переходной платы
    const n = nums.length, pins = {};
    if (n === 1) pins[nums[0]] = [0, 0];
    else if (n === 2) { pins[nums[0]] = [-20, 0]; pins[nums[1]] = [20, 0]; }
    else if (n === 3) nums.forEach((k, i) => { pins[k] = [(i - 1) * 10, 20]; });
    else {
      const half = Math.ceil(n / 2), x0 = -Math.floor((half - 1) / 2) * 10;
      nums.forEach((k, i) => { pins[k] = i < half ? [x0 + i * 10, 15] : [x0 + (n - 1 - i) * 10, -15]; });
    }
    return { pins, fromFootprint: false };
  }

  function kindOf(part, prefix, n) {
    const t = (part.title + ' ' + (part.tags || []).join(' ') + ' ' + ((part.dataStr.head.c_para || {}).package || '')).toLowerCase();
    if (n === 2 && /led/.test(t) && !/driver|strip/.test(t)) return 'led';
    if (n === 2 && prefix === 'R') return 'resistor';
    if (n === 2 && prefix === 'C') return /cp|elec|radial|bd\d|cap-th_bd/i.test(t) ? 'elec' : 'cap';
    if (n === 2 && prefix === 'D') return 'diode';
    if (/^(rv|vr)$/i.test(prefix) || /potentiometer|res-adj/.test(t)) return 'pot';
    if (prefix === 'Q' && n === 3) return 'transistor';
    return 'generic';
  }

  function makeVisual(part, prefix, sym) {
    const nums = Object.keys(sym.pins).sort((a, b) => (+a - +b) || a.localeCompare(b));
    const lay = pinLayout(nums, footprintPads(part.packageDetail && part.packageDetail.dataStr));
    const pins = lay.pins;
    const pts = Object.values(pins);
    const xs = pts.map(p => p[0]), ys = pts.map(p => p[1]);
    const minX = Math.min(...xs), maxX = Math.max(...xs), minY = Math.min(...ys), maxY = Math.max(...ys);
    const kind = kindOf(part, prefix, nums.length);

    const span = maxX - minX;

    if (horiz && span >= 30 && (kind === 'resistor' || kind === 'diode' || kind === 'generic')) {
      // аксиальный корпус между выводами
      const bw = Math.min(36, span - 12), y = minY;

      draw = c => {
        const x1 = minX + (span - bw) / 2, x2 = x1 + bw;
        let s = D.leg(minX, y, x1 + 2, y) + D.leg(x2 - 2, y, maxX, y);
        if (kind === 'resistor') {
          const b = D.resistorBands(c.value);
          s += `<rect x="${x1}" y="${y - 6.5}" width="${bw}" height="13" rx="6" fill="#e3c48f" stroke="#a8885a"/>`;
          [0.2, 0.36, 0.52].forEach((f, i) => { s += `<rect x="${x1 + bw * f}" y="${y - 6.5}" width="3.5" height="13" fill="${b[i]}"/>`; });
          s += `<rect x="${x1 + bw * 0.78}" y="${y - 6.5}" width="3" height="13" fill="#c9a227"/>`;
        } else if (kind === 'diode') {

        } else {

        }
        return s;
      };
    } else if (horiz && (kind === 'led' || kind === 'cap' || kind === 'elec' || kind === 'generic' || kind === 'resistor' || kind === 'diode')) {
      // радиальный корпус над выводами
      const cx = (minX + maxX) / 2, y = minY, r = Math.max(11, span / 2 + 3);
      box = [cx - r - 2, y - 47, 2 * r + 4, 47]; hit = [cx - r, y - 47, 2 * r, 30];
      draw = c => {
        let s = D.leg(minX, y, cx - 5, y - 18) + D.leg(maxX, y, cx + 5, y - 18);
        if (kind === 'led') {
          const col = D.LED_COLORS[(/(red|green|blue|yellow|white|orange)/.exec((part.title + ' ' + (c.value || '')).toLowerCase()) || [])[1] || 'red'];
          s += `<rect x="${cx - 13}" y="${y - 22}" width="26" height="5" rx="2" fill="${col}" stroke="rgba(0,0,0,.35)"/>` +
            `<path d="M ${cx - 11} ${y - 22} L ${cx - 11} ${y - 34} A 11 11 0 0 1 ${cx + 11} ${y - 34} L ${cx + 11} ${y - 22} Z" fill="${col}" fill-opacity=".92" stroke="rgba(0,0,0,.35)"/>`;
        } else if (kind === 'elec') {
          s += `<rect x="${cx - 13}" y="${y - 47}" width="26" height="30" rx="4" fill="#1f3f8a" stroke="#122656"/><rect x="${cx + 5}" y="${y - 47}" width="6" height="30" fill="#c9d3e6"/>`;
        } else {

        (kind === 'pot' ? `<circle cx="${(minX + maxX) / 2}" cy="${y - 22}" r="8" fill="#eceff1"/>` : D.txt((minX + maxX) / 2, y - 22, label, 4.5, '#ddd'));
    } else {
      // корпус между рядами выводов (DIP, модуль на переходной плате)
      const x1 = minX - 7, w = span + 14, h = maxY - minY;
      const body = lay.fromFootprint ? '#2b2b2b' : '#1d6fb8';
      box = [x1, minY - 2, w, h + 4]; hit = [x1, minY + 4, w, h - 8];
      draw = () => {
        let s = pts.map(([x, y]) => `<rect x="${x - 2}" y="${Math.min(y, (minY + maxY) / 2)}" width="4" height="${Math.abs(y - (minY + maxY) / 2)}" fill="#b9bec4"/>`).join('');
        s += `<rect x="${x1}" y="${minY + 4}" width="${w}" height="${h - 8}" rx="2" fill="${body}" stroke="#000"/>`;

        return s;
      };
    }
    return { box, hit, pins, draw, fromFootprint: lay.fromFootprint };
  }

// Построение списка цепей (netlist) из проводов режима «Монтаж»
window.AK = window.AK || {};

AK.POWER_RE = /^(GND|AGND|DGND|PGND|VCC|VDD|VEE|VSS|VIN|VBAT|[+-]\d+(\.\d+)?V\d*)$/i;

AK.buildNetlist = function (project) {
  const parent = new Map();
  const find = k => {
    while (parent.get(k) !== k) { parent.set(k, parent.get(parent.get(k))); k = parent.get(k); }
    return k;
  };
  const union = (a, b) => { a = find(a); b = find(b); if (a !== b) parent.set(a, b); };
  const key = (c, p) => c + ':' + p;

  const comps = new Map(project.components.map(c => [c.id, c]));
  project.components.forEach(c => {
    Object.keys(AK.LIB[c.type].visual.pins).forEach(p => parent.set(key(c.id, p), key(c.id, p)));
  });

  const wired = new Set();
  project.wires.forEach(w => {
    const a = key(w.from.c, w.from.p), b = key(w.to.c, w.to.p);
    if (!parent.has(a) || !parent.has(b)) return;
    union(a, b);
    wired.add(a); wired.add(b);
  });

  // Полоски макетной платы
  project.components.forEach(c => {
    const g = AK.LIB[c.type].groups;
    if (g) g.forEach(ps => ps.forEach(p => union(key(c.id, ps[0]), key(c.id, p))));
  });
  // Выводы в одной точке соединены: ножка в отверстии платы или ножка к ножке
  const at = new Map();
  project.components.forEach(c => Object.keys(AK.LIB[c.type].visual.pins).forEach(p => {
    const s = AK.pinWorld(c, p).map(Math.round).join(',');
    if (!at.has(s)) at.set(s, []);
    at.get(s).push(key(c.id, p));
  }));
  const plugged = new Set();
  at.forEach(ks => {
    if (ks.length < 2) return;
    const legs = ks.filter(k => !AK.LIB[comps.get(k.split(':')[0]).type].board).length;
    // отверстия наложенных друг на друга плат без ножки в них не соединяются
    if (!legs) return;
    ks.forEach(k => { union(ks[0], k); plugged.add(k); });
    // ножка в пустом отверстии ещё не соединение; две ножки в одной точке — соединение
    if (legs > 1) ks.forEach(k => wired.add(k));
  });

  // Имена цепей: 3 — задано пользователем на проводе, 2 — источник питания, 1 — подсказка вывода (GND у Arduino и т.п.)
  const names = new Map(); // root -> [{name, prio}]
  const addName = (k, name, prio) => {
    if (!name) return;
    const r = find(k);
    if (!names.has(r)) names.set(r, []);
    names.get(r).push({ name: String(name).trim(), prio });
  };
  project.wires.forEach(w => { if (w.net && w.net.trim()) addName(key(w.from.c, w.from.p), w.net, 3); });
  project.components.forEach(c => {
    const d = AK.LIB[c.type];
    if (d.powerSource) addName(key(c.id, '1'), d.fixedNet || (c.props && c.props.net) || c.value, 2);
    const pp = Object.assign({}, d.pinPower, d.pinPowerFn && d.pinPowerFn(c));
    Object.entries(pp).forEach(([p, n]) => addName(key(c.id, p), n, 1));
  });

  const warnings = [];
  const groups = new Map();
  parent.forEach((_, k) => {
    const r = find(k);
    if (!groups.has(r)) groups.set(r, []);
    groups.get(r).push(k);
  });

  // Сливаем группы с одинаковыми именами (одноимённые цепи питания соединены глобально)
  const byName = new Map();
  const nets = [];
  groups.forEach((keys, r) => {
    const cand = names.get(r) || [];
    let name = null, prio = 0, power = false;
    if (cand.length) {
      prio = Math.max(...cand.map(c => c.prio));
      const top = [...new Set(cand.filter(c => c.prio === prio).map(c => c.name))];
      name = top[0];
      if (top.length > 1) {
        const pins = keys.filter(k => wired.has(k)).length;
        if (pins > 0 || prio > 1) warnings.push({ level: prio >= 2 ? 'error' : 'warn', text: `Замкнуты разные цепи: ${top.join(' и ')}` });
      }
      // источник питания — всегда шина; имя, заданное на проводе, — только если похоже на питание
      power = cand.some(c => c.prio === 2) || AK.POWER_RE.test(name);
    }
    const pins = keys.map(k => { const [c, p] = k.split(':'); return { c, p }; });
    const net = { name, prio, power, pins, wired: keys.some(k => wired.has(k)) };
    if (name && byName.has(name)) {
      const other = byName.get(name);
      other.pins.push(...pins);
      other.prio = Math.max(other.prio, prio);
      other.power = other.power || power;
      other.wired = other.wired || net.wired;
    } else {
      if (name) byName.set(name, net);
      nets.push(net);
    }
  });

  let auto = 1;
  const used = new Set(nets.filter(n => n.name).map(n => n.name));
  const pinNet = new Map();
  nets.forEach((n, i) => {
    n.id = i;
    // реальные (не виртуальные) выводы
    n.real = n.pins.filter(p => !AK.LIB[comps.get(p.c).type].virtual);
    // Цепь, названная лишь подсказкой вывода (GND у Arduino) и ни с чем не соединённая, не рисуется
    const hintOnly = n.prio === 1 && !n.wired;
    n.drawn = !hintOnly && (n.real.length >= 2 || (n.power && n.real.length >= 1));
    if (!n.name && n.drawn) {
      while (used.has('N' + auto)) auto++;
      n.name = 'N' + auto++;
    }
    n.pins.forEach(p => pinNet.set(key(p.c, p.p), n));
  });

  // Предупреждения о неподключённых выводах и типичных ошибках (как ERC в KiCad)
  project.components.forEach(c => {
    const d = AK.LIB[c.type];
    if (d.virtual) return;
    if (d.sym) Object.entries(d.sym.pins).forEach(([p, sp]) => {
      const n = pinNet.get(key(c.id, p));
      if (!n || !n.drawn) return;
      if (sp.type === 'pwr' && !n.power) warnings.push({ level: 'warn', text: `${c.ref}: вывод питания ${sp.name} подключён к сигнальной цепи ${n.name}` });
      if ((sp.type === 'out' || sp.type === 'oc') && n.power) warnings.push({ level: 'warn', text: `${c.ref}: выход ${sp.name} замкнут на шину ${n.name}` });
    });
    const pins = Object.keys(d.visual.pins);
    const free = pins.filter(p => { const n = pinNet.get(key(c.id, p)); return !n || !n.drawn; });
    if (free.length === pins.length) warnings.push({ level: 'info', text: `${c.ref}: не подключён` });
    else if (free.length && pins.length <= 3) warnings.push({ level: 'info', text: `${c.ref}: свободный вывод ${free.map(p => AK.pinName(c.type, p)).join(', ')}` });
  });

  return { nets, pinNet, warnings, key, plugged, drawn: nets.filter(n => n.drawn) };
};

// Автоматическая генерация принципиальной схемы из netlist.
// Координаты в единицах U (100 mil), ось Y вниз, все выводы на целой сетке.
window.AK = window.AK || {};

AK.OUT = { R: [-1, 0], L: [1, 0], U: [0, 1], D: [0, -1] }; // dir вывода -> направление «наружу»
AK.isGndName = n => /^(GND|AGND|DGND|PGND|VSS|VEE)$/i.test(n) || /^-/.test(n);

AK.generateSchematic = function (project, nl) {
  const comps = project.components.filter(c => !AK.LIB[c.type].virtual);
  const sym = c => AK.LIB[c.type].sym;
  const K = (c, p) => c.id + ':' + p;
  const netOf = (c, p) => { const n = nl.pinNet.get(K(c, p)); return n && n.drawn ? n : null; };

  // ---------------- Расстановка ----------------
  const MX = 2.5, MY = 1.5;
  const pos = new Map();
  const rects = [];
  const rectAt = (c, x, y) => { const b = sym(c).bbox; return { x1: x + b.x1, y1: y + b.y1, x2: x + b.x2, y2: y + b.y2 }; };
  const overlaps = r => rects.some(q => r.x1 - MX < q.x2 + MX && r.x2 + MX > q.x1 - MX && r.y1 - MY < q.y2 + MY && r.y2 + MY > q.y1 - MY);
  const put = (c, x, y) => { pos.set(c.id, { x, y }); rects.push(rectAt(c, x, y)); };

  // сигнальные цепи компонента (цепи питания на расстановку не влияют)
  const sigNets = new Map(comps.map(c => [c.id, new Set(Object.keys(sym(c).pins).map(p => netOf(c, p)).filter(n => n && !n.power))]));
  const weight = (a, b) => { let w = 0; sigNets.get(a.id).forEach(n => { if (sigNets.get(b.id).has(n)) w++; }); return w; };

  comps.forEach(c => { if (c.sch && isFinite(c.sch.x)) put(c, Math.round(c.sch.x), Math.round(c.sch.y)); });
  let rest = comps.filter(c => !pos.has(c.id));

  function cost(c, x, y) {
    let s = 0;
    const s0 = sym(c);
    Object.entries(s0.pins).forEach(([p, pp]) => {
      const n = netOf(c, p);
      if (!n || n.power) return;
      let best = Infinity;
      n.real.forEach(q => {
        if (q.c === c.id || !pos.has(q.c)) return;
        const oc = comps.find(o => o.id === q.c), op = pos.get(q.c), qp = sym(oc).pins[q.p];
        best = Math.min(best, Math.abs(x + pp.x - op.x - qp.x) + Math.abs(y + pp.y - op.y - qp.y));
      });
      if (best < Infinity) s += best;
    });
    return s;
  }

  while (rest.length) {
    // следующий — самый связанный с уже поставленными (или вообще самый связанный)
    const placedList = comps.filter(c => pos.has(c.id));
    let pick = null, pw = -1;
    rest.forEach(c => {
      let w = placedList.length ? placedList.reduce((a, o) => a + weight(c, o), 0) * 100 : 0;
      w += rest.reduce((a, o) => a + (o !== c ? weight(c, o) : 0), 0) + Object.keys(sym(c).pins).length * 0.01;
      if (w > pw) { pw = w; pick = c; }
    });
    const c = pick;
    rest = rest.filter(o => o !== c);

    // центр поиска — среднее положение связанных выводов
    let cx = 0, cy = 0, cnt = 0;
    placedList.forEach(o => { if (weight(c, o) > 0) { const p = pos.get(o.id); cx += p.x; cy += p.y; cnt++; } });
    if (!cnt && placedList.length) {
      placedList.forEach(o => { const p = pos.get(o.id); cx += p.x; cy += p.y; cnt++; });
      // несвязанные — справа от всего остального
      const maxX = Math.max(...rects.map(r => r.x2));
      cx = maxX + 4 - sym(c).bbox.x1; cy = cy / cnt; cnt = 1;
    }
    if (cnt) { cx = Math.round(cx / cnt); cy = Math.round(cy / cnt); }

    let best = null, bestCost = Infinity, firstR = -1;
    for (let r = 0; r < 80; r++) {
      if (firstR >= 0 && r > firstR + 6) break;
      const ring = [];
      if (r === 0) ring.push([0, 0]);
      else {
        for (let i = -r; i <= r; i++) { ring.push([i, -r], [i, r]); }
        for (let i = -r + 1; i <= r - 1; i++) { ring.push([-r, i], [r, i]); }
      }
      ring.forEach(([dx, dy]) => {
        const x = cx + dx, y = cy + dy;
        if (overlaps(rectAt(c, x, y))) return;
        if (firstR < 0) firstR = r;
        const k = cost(c, x, y) + 0.15 * (Math.abs(dx) + Math.abs(dy));
        if (k < bestCost) { bestCost = k; best = [x, y]; }
      });
    }
    put(c, best[0], best[1]);
  }

  // ---------------- Трассировка ----------------
  const segs = [];   // {x1,y1,x2,y2,net}
  const pts = [];    // {x,y,net}
  const obst = comps.map(c => { const p = pos.get(c.id), r = rectAt(c, p.x, p.y); return { x1: r.x1 - 0.5, y1: r.y1 - 0.5, x2: r.x2 + 0.5, y2: r.y2 + 0.5 }; });
  const labels = [], power = [], ncs = [], wires = [];

  const onSeg = (x, y, s) => (s.x1 === s.x2 ? (x === s.x1 && y >= Math.min(s.y1, s.y2) && y <= Math.max(s.y1, s.y2))
    : (y === s.y1 && x >= Math.min(s.x1, s.x2) && x <= Math.max(s.x1, s.x2)));
  // отрезок проходит через внутренность прямоугольника
  const segHitsRect = (s, r) => Math.max(s.x1, s.x2) > r.x1 && Math.min(s.x1, s.x2) < r.x2 &&
    Math.max(s.y1, s.y2) > r.y1 && Math.min(s.y1, s.y2) < r.y2;

  function segOk(s, net, extraRects) {
    if (s.x1 === s.x2 && s.y1 === s.y2) return true;
    for (const r of obst) if (segHitsRect(s, r)) return false;
    if (extraRects) for (const r of extraRects) if (r.net !== net && segHitsRect(s, r)) return false;
    for (const p of pts) if (p.net !== net && onSeg(p.x, p.y, s)) return false;
    for (const o of segs) {
      if (o.net === net) continue;
      const sv = s.x1 === s.x2, ov = o.x1 === o.x2;
      if (sv === ov) {
        // коллинеарные отрезки разных цепей не должны касаться
        if (sv && s.x1 === o.x1 && Math.max(Math.min(s.y1, s.y2), Math.min(o.y1, o.y2)) <= Math.min(Math.max(s.y1, s.y2), Math.max(o.y1, o.y2))) return false;
        if (!sv && s.y1 === o.y1 && Math.max(Math.min(s.x1, s.x2), Math.min(o.x1, o.x2)) <= Math.min(Math.max(s.x1, s.x2), Math.max(o.x1, o.x2))) return false;
      } else {
        // пересечение допускается, а Т-образное касание — нет (это было бы соединение)
        const [vx, vy1, vy2, hy, hx1, hx2] = sv
          ? [s.x1, Math.min(s.y1, s.y2), Math.max(s.y1, s.y2), o.y1, Math.min(o.x1, o.x2), Math.max(o.x1, o.x2)]
          : [o.x1, Math.min(o.y1, o.y2), Math.max(o.y1, o.y2), s.y1, Math.min(s.x1, s.x2), Math.max(s.x1, s.x2)];
        if (vx >= hx1 && vx <= hx2 && hy >= vy1 && hy <= vy2) {
          const isEnd = (x, y, q) => (q.x1 === x && q.y1 === y) || (q.x2 === x && q.y2 === y);
          if (isEnd(vx, hy, s) || isEnd(vx, hy, o)) return false;
        }
      }
    }
    return true;
  }

  const addSeg = (x1, y1, x2, y2, net) => {
    if (x1 === x2 && y1 === y2) return;
    const s = { x1, y1, x2, y2, net };
    segs.push(s);
    wires.push(s);
  };

  // 1. Отводы (стабы) от каждого подключённого вывода
  const stubs = new Map(); // net -> [{x,y,out,pin:[x,y],comp}]
  comps.forEach(c => {
    const p0 = pos.get(c.id);
    Object.entries(sym(c).pins).forEach(([p, pp]) => {
      const px = p0.x + pp.x, py = p0.y + pp.y;
      const n = netOf(c, p);
      if (!n) { ncs.push([px, py]); pts.push({ x: px, y: py, net: -1 }); return; }
      const out = AK.OUT[pp.dir];
      pts.push({ x: px, y: py, net: n.id });
      const st = { x: px + out[0], y: py + out[1], out, px, py, comp: c, cx: p0.x };
      if (!stubs.has(n)) stubs.set(n, []);
      stubs.get(n).push(st);
    });
  });
  stubs.forEach((list, n) => list.forEach(st => {
    addSeg(st.px, st.py, st.x, st.y, n.id);
    pts.push({ x: st.x, y: st.y, net: n.id });
  }));

  // 2. Символы питания
  const pwrRects = [];
  // место под возможную метку у каждого сигнального отвода — символы питания его не занимают
  const labelZones = [];
  stubs.forEach((list, n) => {
    if (n.power) return;
    const L = 0.6 + 0.5 * String(n.name || 'N1').length;
    list.forEach(st => {
      const o = st.out, ex = st.x + o[0] * L, ey = st.y + o[1] * L;
      labelZones.push({ x1: Math.min(st.x, ex) - (o[0] ? 0 : 0.45), x2: Math.max(st.x, ex) + (o[0] ? 0 : 0.45), y1: Math.min(st.y, ey) - (o[1] ? 0 : 0.45), y2: Math.max(st.y, ey) + (o[1] ? 0 : 0.45) });
    });
  });
  const hit = (r, q) => r.x1 < q.x2 && r.x2 > q.x1 && r.y1 < q.y2 && r.y2 > q.y1;
  const regionFree = (r, net, zones) => {
    for (const o of obst) if (hit(r, o)) return false;
    for (const q of pwrRects) if (hit(r, q)) return false;
    if (zones) for (const q of labelZones) if (hit(r, q)) return false;
    for (const p of pts) if (p.net !== net && p.x >= r.x1 && p.x <= r.x2 && p.y >= r.y1 && p.y <= r.y2) return false;
    for (const s of segs) if (s.net !== net && segHitsRect(s, r)) return false;
    return true;
  };
  stubs.forEach((list, n) => {
    if (!n.power) return;
    const gnd = AK.isGndName(n.name);
    list.forEach(st => {
      // область символа вместе с подписью (+9V, VCC… над чертой)
      const tw = Math.max(0.7, 0.3 * String(n.name).length);
      const reg = (x, y) => gnd ? { x1: x - 0.8, y1: y, x2: x + 0.8, y2: y + 1.4 } : { x1: x - tw, y1: y - 1.9, x2: x + tw, y2: y };
      const against = gnd ? st.out[1] < 0 : st.out[1] > 0; // символ «смотрит» обратно на вывод
      const side = st.px >= st.cx ? 1 : -1;
      let ok = null;
      // сначала — не занимая места будущих меток, затем без этого ограничения
      for (const zones of [true, false]) {
        for (let t = 0; t <= 6 && !ok; t++) {
          let path;
          if (!against) path = [[st.x, st.y], [st.x + st.out[0] * t, st.y + st.out[1] * t]];
          else path = [[st.x, st.y], [st.x, st.y], [st.x + side * (2 + t), st.y]];
          const end = path[path.length - 1];
          const r = reg(end[0], end[1]);
          let good = regionFree(r, n.id, zones);
          for (let i = 0; good && i < path.length - 1; i++) {
            good = segOk({ x1: path[i][0], y1: path[i][1], x2: path[i + 1][0], y2: path[i + 1][1] }, n.id, pwrRects);
          }
          if (good) ok = { path, r };
        }
        if (ok) break;
      }
      if (!ok) { // запасной вариант — как есть
        ok = { path: [[st.x, st.y]], r: reg(st.x, st.y) };
      }
      for (let i = 0; i < ok.path.length - 1; i++) addSeg(ok.path[i][0], ok.path[i][1], ok.path[i + 1][0], ok.path[i + 1][1], n.id);
      const e = ok.path[ok.path.length - 1];
      pts.push({ x: e[0], y: e[1], net: n.id });
      ok.r.net = n.id;
      pwrRects.push(ok.r);
      power.push({ x: e[0], y: e[1], name: n.name, kind: gnd ? 'gnd' : 'vcc' });
    });
  });

  // 3. Сигнальные цепи: минимальное остовное дерево + Г/Z-образные трассы, иначе — метки
  const dirOf = out => out[0] > 0 ? 'R' : out[0] < 0 ? 'L' : out[1] < 0 ? 'U' : 'D';
  const labeled = new Set();
  const addLabel = (st, n) => {
    const k = st.x + ',' + st.y;
    if (labeled.has(k)) return;
    labeled.add(k);
    labels.push({ x: st.x, y: st.y, dir: dirOf(st.out), text: n.name, net: n.id });
  };

  function candidates(a, b) {
    const list = [];
    if (a.x === b.x || a.y === b.y) list.push([[a.x, a.y], [b.x, b.y]]);
    list.push([[a.x, a.y], [b.x, a.y], [b.x, b.y]]);
    list.push([[a.x, a.y], [a.x, b.y], [b.x, b.y]]);
    const xs = [], ys = [];
    const lo = (u, v) => Math.min(u, v), hi = (u, v) => Math.max(u, v);
    for (let x = lo(a.x, b.x) + 1; x < hi(a.x, b.x); x++) xs.push(x);
    for (let y = lo(a.y, b.y) + 1; y < hi(a.y, b.y); y++) ys.push(y);
    const mid = (arr, m) => arr.sort((p, q) => Math.abs(p - m) - Math.abs(q - m));
    mid(xs, (a.x + b.x) / 2).forEach(x => list.push([[a.x, a.y], [x, a.y], [x, b.y], [b.x, b.y]]));
    mid(ys, (a.y + b.y) / 2).forEach(y => list.push([[a.x, a.y], [a.x, y], [b.x, y], [b.x, b.y]]));
    for (let k = 1; k <= 4; k++) {
      [lo(a.x, b.x) - k, hi(a.x, b.x) + k].forEach(x => list.push([[a.x, a.y], [x, a.y], [x, b.y], [b.x, b.y]]));
      [lo(a.y, b.y) - k, hi(a.y, b.y) + k].forEach(y => list.push([[a.x, a.y], [a.x, y], [b.x, y], [b.x, b.y]]));
    }
    return list;
  }

  const signalNets = [...stubs.keys()].filter(n => !n.power).sort((p, q) => stubs.get(p).length - stubs.get(q).length);
  signalNets.forEach(n => {
    const list = stubs.get(n);
    if (list.length < 2) { list.forEach(st => addLabel(st, n)); return; }
    // Прим
    const inTree = [0], edges = [];
    while (inTree.length < list.length) {
      let best = null;
      inTree.forEach(i => list.forEach((b, j) => {
        if (inTree.includes(j)) return;
        const a = list[i], d = Math.abs(a.x - b.x) + Math.abs(a.y - b.y);
        if (!best || d < best.d) best = { i, j, d };
      }));
      inTree.push(best.j);
      edges.push(best);
    }
    edges.forEach(e => {
      const a = list[e.i], b = list[e.j];
      let route = null;
      if (e.d <= 30) {
        for (const cand of candidates(a, b)) {
          let good = true;
          for (let i = 0; good && i < cand.length - 1; i++) {
            good = segOk({ x1: cand[i][0], y1: cand[i][1], x2: cand[i + 1][0], y2: cand[i + 1][1] }, n.id, pwrRects);
          }
          if (good) { route = cand; break; }
        }
      }
      if (route) {
        for (let i = 0; i < route.length - 1; i++) addSeg(route[i][0], route[i][1], route[i + 1][0], route[i + 1][1], n.id);
        route.forEach(([x, y]) => pts.push({ x, y, net: n.id }));
      } else {
        addLabel(a, n); addLabel(b, n);
      }
    });
  });

  // 4. Точки соединения: там, где сходятся 3+ конца провода
  const ends = new Map();
  wires.forEach(s => [[s.x1, s.y1], [s.x2, s.y2]].forEach(([x, y]) => { const k = x + ',' + y; ends.set(k, (ends.get(k) || 0) + 1); }));
  const pinSet = new Set();
  comps.forEach(c => { const p0 = pos.get(c.id); Object.values(sym(c).pins).forEach(pp => pinSet.add((p0.x + pp.x) + ',' + (p0.y + pp.y))); });
  const junctions = [];
  ends.forEach((cnt, k) => { if (cnt + (pinSet.has(k) ? 1 : 0) >= 3) junctions.push(k.split(',').map(Number)); });

  // габариты
  let bx1 = Infinity, by1 = Infinity, bx2 = -Infinity, by2 = -Infinity;
  const grow = (x1, y1, x2, y2) => { bx1 = Math.min(bx1, x1); by1 = Math.min(by1, y1); bx2 = Math.max(bx2, x2); by2 = Math.max(by2, y2); };
  rects.forEach(r => grow(r.x1, r.y1 - 1.5, r.x2, r.y2 + 1.5));
  wires.forEach(s => grow(Math.min(s.x1, s.x2), Math.min(s.y1, s.y2), Math.max(s.x1, s.x2), Math.max(s.y1, s.y2)));
  pwrRects.forEach(r => grow(r.x1, r.y1 - 0.8, r.x2, r.y2 + 0.8));
  labels.forEach(l => grow(l.x - 4, l.y - 4, l.x + 4, l.y + 4));
  if (!isFinite(bx1)) { bx1 = 0; by1 = 0; bx2 = 10; by2 = 10; }

  return {
    symbols: comps.map(c => ({ comp: c, def: AK.LIB[c.type], x: pos.get(c.id).x, y: pos.get(c.id).y })),
    wires, labels, power, junctions, ncs,
    bbox: { x1: bx1, y1: by1, x2: bx2, y2: by2 },
    nets: nl.drawn,
  };
};

AK.symbolSVG = function (def, comp, showText) {
  const s = def.sym, e = AK.util.esc;
  let out = '';
  s.gfx.forEach(g => {
    const w = g.w ? ` style="stroke-width:${0.09 * g.w}"` : '';
    if (g.t === 'line') out += `<polyline class="s-body" points="${g.pts.map(p => p.join(',')).join(' ')}"${w}/>`;
    else if (g.t === 'rect') out += `<rect class="s-body ${g.fill === 'body' ? 's-fill' : ''}" x="${g.x1}" y="${g.y1}" width="${g.x2 - g.x1}" height="${g.y2 - g.y1}"/>`;
    else if (g.t === 'circle') out += `<circle class="s-body" cx="${g.cx}" cy="${g.cy}" r="${g.r}"/>`;
    else if (g.t === 'poly') out += `<polygon class="s-body s-solid" points="${g.pts.map(p => p.join(',')).join(' ')}"/>`;
    else if (g.t === 'text') out += `<text class="s-gtext" x="${g.x}" y="${g.y}" font-size="${g.size}" text-anchor="middle" dominant-baseline="central">${e(g.text)}</text>`;
  });
  Object.entries(s.pins).forEach(([num, p]) => {
    const [dx, dy] = AK.DIRV[p.dir];
    const ex = p.x + dx * p.len, ey = p.y + dy * p.len;
    out += `<line class="s-pin" x1="${p.x}" y1="${p.y}" x2="${ex}" y2="${ey}"/>`;
    if (s.showNames && p.name) {
      // у вертикального вывода имя — над/под концом вывода, внутри корпуса
      const anchor = dy ? 'middle' : p.dir === 'R' ? 'start' : 'end';
      out += `<text class="s-pinname" x="${ex + dx * 0.25}" y="${ey + dy * 0.4}" text-anchor="${anchor}" dominant-baseline="central">${e(p.name)}</text>`;
    }
    if (s.showNums) {
      out += `<text class="s-pinnum" x="${(p.x + ex) / 2}" y="${p.y - 0.12}" text-anchor="middle">${e(num)}</text>`;
    }
  });
  if (showText !== false && comp) {
    out += `<text class="s-ref" x="${s.ref[0]}" y="${s.ref[1]}" text-anchor="${s.ref[2]}" dominant-baseline="central">${e(comp.ref)}</text>`;
    if (comp.value && comp.value !== comp.ref)
      out += `<text class="s-val" x="${s.val[0]}" y="${s.val[1]}" text-anchor="${s.val[2]}" dominant-baseline="central">${e(comp.value)}</text>`;
  }
  return out;
};

AK.powerSVG = function (p) {
  const { x, y } = p, e = AK.util.esc;
  if (p.kind === 'gnd') {
    return `<path class="s-pwr" d="M${x} ${y}V${y + 0.5}M${x - 0.6} ${y + 0.5}H${x + 0.6}M${x - 0.37} ${y + 0.75}H${x + 0.37}M${x - 0.14} ${y + 1}H${x + 0.14}"/>` +
      (p.name !== 'GND' ? `<text class="s-pwrtext" x="${x}" y="${y + 1.55}" text-anchor="middle">${e(p.name)}</text>` : '');
  }
  return `<path class="s-pwr" d="M${x} ${y}V${y - 0.6}M${x - 0.5} ${y - 0.6}H${x + 0.5}"/>` +
    `<text class="s-pwrtext" x="${x}" y="${y - 0.85}" text-anchor="middle">${e(p.name)}</text>`;
};

AK.labelSVG = function (l) {
  const e = AK.util.esc, o = 0.12;
  if (l.dir === 'R') return `<text class="s-label" x="${l.x + o}" y="${l.y - o}" text-anchor="start">${e(l.text)}</text>`;
  if (l.dir === 'L') return `<text class="s-label" x="${l.x - o}" y="${l.y - o}" text-anchor="end">${e(l.text)}</text>`;
  if (l.dir === 'U') return `<text class="s-label" x="${l.x - o}" y="${l.y - o}" text-anchor="start" transform="rotate(-90 ${l.x - o} ${l.y - o})">${e(l.text)}</text>`;
  return `<text class="s-label" x="${l.x - o}" y="${l.y + o}" text-anchor="end" transform="rotate(-90 ${l.x - o} ${l.y + o})">${e(l.text)}</text>`;
};

AK.schematicSVG = function (m, opts) {
  opts = opts || {};
  let s = '';
  m.wires.forEach(w => { s += `<line class="s-wire" data-net="${w.net}" x1="${w.x1}" y1="${w.y1}" x2="${w.x2}" y2="${w.y2}"/>`; });
  m.junctions.forEach(([x, y]) => { s += `<circle class="s-junc" cx="${x}" cy="${y}" r=".17"/>`; });
  m.ncs.forEach(([x, y]) => { s += `<path class="s-nc" d="M${x - 0.3} ${y - 0.3}L${x + 0.3} ${y + 0.3}M${x + 0.3} ${y - 0.3}L${x - 0.3} ${y + 0.3}"/>`; });
  m.power.forEach(p => { s += AK.powerSVG(p); });
  m.labels.forEach(l => { s += AK.labelSVG(l); });
  m.symbols.forEach(sy => {
    const b = sy.def.sym.bbox;
    s += `<g class="sym" data-id="${sy.comp.id}" transform="translate(${sy.x} ${sy.y})">` +
      `<rect class="sym-hit" x="${b.x1 - 0.3}" y="${b.y1 - 0.3}" width="${b.x2 - b.x1 + 0.6}" height="${b.y2 - b.y1 + 0.6}" fill="transparent"/>` +
      AK.symbolSVG(sy.def, sy.comp) + `</g>`;
  });
  return s;
};

AK.schematicStandaloneSVG = function (m, title) {
  const pad = 2, b = m.bbox;
  const w = b.x2 - b.x1 + pad * 2, h = b.y2 - b.y1 + pad * 2;
  const px = 40;
  return `<?xml version="1.0" encoding="UTF-8"?>\n<svg xmlns="http://www.w3.org/2000/svg" width="${w * px}" height="${h * px}" viewBox="${b.x1 - pad} ${b.y1 - pad} ${w} ${h}">` +
    `<title>${AK.util.esc(title || 'Схема')}</title><style>${AK.SCH_STYLE}</style>` +
    `<rect x="${b.x1 - pad}" y="${b.y1 - pad}" width="${w}" height="${h}" fill="#fff"/>${AK.schematicSVG(m)}</svg>`;
};

AK.SchView = class {
  constructor(svg, app) {
    this.svg = svg; this.app = app;
    this.view = { x: 60, y: 60, z: 34 };
    this.model = null;
    svg.innerHTML = `<defs><pattern id="schgrid" width="1" height="1" patternUnits="userSpaceOnUse"><circle cx="0" cy="0" r=".045" fill="#c4c9d1"/></pattern><style>${AK.SCH_STYLE}</style></defs>` +
      `<g class="root"><rect class="bg" x="-500" y="-500" width="1000" height="1000" fill="url(#schgrid)"/><g class="content"></g></g>`;
    this.root = svg.querySelector('.root');
    this.content = svg.querySelector('.content');
    this.bind();
  }

  toWorld(e) {
    const r = this.svg.getBoundingClientRect();
    return [(e.clientX - r.left - this.view.x) / this.view.z, (e.clientY - r.top - this.view.y) / this.view.z];
  }
  applyView() { this.root.setAttribute('transform', `translate(${this.view.x} ${this.view.y}) scale(${this.view.z})`); }

  render(model) {
    this.model = model;
    this.content.innerHTML = AK.schematicSVG(model);
    this.applyView();
  }

  fit() {
    if (!this.model) return;
    const b = this.model.bbox, r = this.svg.getBoundingClientRect();
    if (!r.width) return;
    const z = Math.min(r.width / (b.x2 - b.x1 + 6), r.height / (b.y2 - b.y1 + 6), 60);
    this.view.z = z;
    this.view.x = r.width / 2 - (b.x1 + b.x2) / 2 * z;
    this.view.y = r.height / 2 - (b.y1 + b.y2) / 2 * z;
    this.applyView();
  }

  zoomBy(f, cx, cy) {
    const r = this.svg.getBoundingClientRect();
    if (cx == null) { cx = r.width / 2; cy = r.height / 2; }
    const nz = Math.max(4, Math.min(200, this.view.z * f));
    this.view.x = cx - (cx - this.view.x) * nz / this.view.z;
    this.view.y = cy - (cy - this.view.y) * nz / this.view.z;
    this.view.z = nz;
    this.applyView();
  }

  bind() {
    const svg = this.svg;
    svg.addEventListener('wheel', e => {
      e.preventDefault();
      const r = svg.getBoundingClientRect();
      this.zoomBy(e.deltaY < 0 ? 1.15 : 1 / 1.15, e.clientX - r.left, e.clientY - r.top);
    }, { passive: false });

    svg.addEventListener('pointerdown', e => {
      if (e.button !== 0 && e.button !== 1) return;
      const g = e.target.closest('.sym');
      svg.setPointerCapture(e.pointerId);
      if (g && e.button === 0) {
        const sy = this.model.symbols.find(s => s.comp.id === g.dataset.id);
        const [wx, wy] = this.toWorld(e);
        this.drag = { sy, g, dx: wx - sy.x, dy: wy - sy.y, moved: false };
        this.app.selectComp(sy.comp.id);
        this.content.querySelectorAll('.sym.sel').forEach(n => n.classList.remove('sel'));
        g.classList.add('sel');
      } else {
        this.pan = { x: e.clientX, y: e.clientY, vx: this.view.x, vy: this.view.y };
      }
    });
    svg.addEventListener('pointermove', e => {
      if (this.drag) {
        const [wx, wy] = this.toWorld(e);
        const nx = Math.round(wx - this.drag.dx), ny = Math.round(wy - this.drag.dy);
        if (nx !== this.drag.sy.x || ny !== this.drag.sy.y) {
          this.drag.moved = true;
          this.drag.sy.x = nx; this.drag.sy.y = ny;
          this.drag.g.setAttribute('transform', `translate(${nx} ${ny})`);
          this.content.classList.add('dragging');
        }
      } else if (this.pan) {
        this.view.x = this.pan.vx + e.clientX - this.pan.x;
        this.view.y = this.pan.vy + e.clientY - this.pan.y;
        this.applyView();
      }
    });
    const up = () => {
      if (this.drag && this.drag.moved) {
        // фиксируем текущее расположение всех символов и перестраиваем провода
        this.app.checkpoint();
        this.model.symbols.forEach(s => { s.comp.sch = { x: s.x, y: s.y }; });
        this.content.classList.remove('dragging');
        this.app.changed({ keepView: true, schOnly: true });
      }
      this.drag = null; this.pan = null;
    };
    svg.addEventListener('pointerup', up);
    svg.addEventListener('pointercancel', up);
  }
};

// ---- js/export-kicad.js ----
// Экспорт в KiCad 8 (.kicad_sch) и netlist (.net)
window.AK = window.AK || {};

AK.exportKicad = (function () {
  const MM = 2.54;
  const f = n => AK.util.fmt(n, 4);
  const q = s => '"' + String(s == null ? '' : s).replace(/\\/g, '\\\\').replace(/"/g, '\\"') + '"';
  const FONT = '(effects (font (size 1.27 1.27)))';
  const FONT_HIDE = '(effects (font (size 1.27 1.27)) hide)';
  const STROKE = '(stroke (width 0.254) (type default))';
  const ANG = { R: 0, L: 180, U: 90, D: 270 }; // dir (к корпусу) -> угол KiCad (ось Y вверх)

  // координаты библиотеки KiCad: Y вверх
  const lx = x => f(x * MM), ly = y => f(-y * MM);

  // libPrefix=true — для lib_symbols внутри схемы, false — для файла библиотеки .kicad_sym
  function libSymbol(def, libPrefix = true) {
    const s = def.sym, name = def.kicadName;
    const out = [];
    const TYPES = { pwr: 'power_in', out: 'output', in: 'input', oc: 'open_collector' };
    const pinType = p => TYPES[p.type] || (s.ic ? 'bidirectional' : 'passive');
    out.push(`(symbol ${q((libPrefix ? 'AutoKiCad:' : '') + name)}` +
      (s.showNums ? '' : ' (pin_numbers hide)') +
      (s.showNames ? ' (pin_names (offset 0.508))' : ' (pin_names (offset 0) hide)') +
      ' (exclude_from_sim no) (in_bom yes) (on_board yes)');
    out.push(`  (property "Reference" ${q(def.prefix)} (at ${lx(s.ref[0])} ${ly(s.ref[1])} 0) ${FONT})`);
    out.push(`  (property "Value" ${q(def.value)} (at ${lx(s.val[0])} ${ly(s.val[1])} 0) ${FONT})`);
    out.push(`  (property "Footprint" ${q(def.footprint || '')} (at 0 0 0) ${FONT_HIDE})`);
    out.push(`  (property "Datasheet" "~" (at 0 0 0) ${FONT_HIDE})`);
    out.push(`  (symbol ${q(name + '_0_1')}`);
    s.gfx.forEach(g => {
      const st = g.w ? `(stroke (width ${f(0.254 * g.w)}) (type default))` : STROKE;
      if (g.t === 'line') out.push(`    (polyline (pts ${g.pts.map(p => `(xy ${lx(p[0])} ${ly(p[1])})`).join(' ')}) ${st} (fill (type none)))`);
      else if (g.t === 'poly') out.push(`    (polyline (pts ${[...g.pts, g.pts[0]].map(p => `(xy ${lx(p[0])} ${ly(p[1])})`).join(' ')}) ${st} (fill (type outline)))`);
      else if (g.t === 'rect') out.push(`    (rectangle (start ${lx(g.x1)} ${ly(g.y1)}) (end ${lx(g.x2)} ${ly(g.y2)}) ${st} (fill (type ${g.fill === 'body' ? 'background' : 'none'})))`);
      else if (g.t === 'circle') out.push(`    (circle (center ${lx(g.cx)} ${ly(g.cy)}) (radius ${f(g.r * MM)}) ${st} (fill (type none)))`);
      else if (g.t === 'text') out.push(`    (text ${q(g.text)} (at ${lx(g.x)} ${ly(g.y)} 0) ${FONT})`);
    });
    out.push('  )');
    out.push(`  (symbol ${q(name + '_1_1')}`);
    Object.entries(s.pins).forEach(([num, p]) => {
      out.push(`    (pin ${pinType(p)} line (at ${lx(p.x)} ${ly(p.y)} ${ANG[p.dir]}) (length ${f(p.len * MM)}) (name ${q(p.name || '~')} ${FONT}) (number ${q(num)} ${FONT}))`);
    });
    out.push('  )');
    out.push(')');
    return out.join('\n    ');
  }

  // символы питания живут в библиотеке AutoKiCad (PWR_<имя>), чтобы не конфликтовать со штатной power
  const pwrId = name => 'PWR_' + name;

  const PWR_FLAG = (libPrefix = true) => `(symbol "${libPrefix ? 'AutoKiCad:' : ''}PWR_FLAG" (power) (pin_numbers hide) (pin_names (offset 0) hide) (exclude_from_sim no) (in_bom yes) (on_board yes)
      (property "Reference" "#FLG" (at 0 1.905 0) ${FONT_HIDE})
      (property "Value" "PWR_FLAG" (at 0 3.81 0) ${FONT_HIDE})
      (property "Footprint" "" (at 0 0 0) ${FONT_HIDE})
      (property "Datasheet" "~" (at 0 0 0) ${FONT_HIDE})
      (symbol "PWR_FLAG_0_0"
        (pin power_out line (at 0 0 90) (length 0) (name "~" ${FONT}) (number "1" ${FONT}))
      )
      (symbol "PWR_FLAG_0_1"
        (polyline (pts (xy 0 0) (xy 0 1.27) (xy -1.016 1.905) (xy 0 2.54) (xy 1.016 1.905) (xy 0 1.27)) ${STROKE} (fill (type none)))
      )
    )`;

  function powerLib(name, gnd, libPrefix = true) {
    const shape = gnd
      ? `(polyline (pts (xy 0 0) (xy 0 -1.27)) ${STROKE} (fill (type none)))
      (polyline (pts (xy -1.524 -1.27) (xy 1.524 -1.27)) ${STROKE} (fill (type none)))
      (polyline (pts (xy -0.9398 -1.905) (xy 0.9398 -1.905)) ${STROKE} (fill (type none)))
      (polyline (pts (xy -0.3556 -2.54) (xy 0.3556 -2.54)) ${STROKE} (fill (type none)))`
      : `(polyline (pts (xy 0 0) (xy 0 1.524)) ${STROKE} (fill (type none)))
      (polyline (pts (xy -1.27 1.524) (xy 1.27 1.524)) ${STROKE} (fill (type none)))`;
    return `(symbol ${q((libPrefix ? 'AutoKiCad:' : '') + pwrId(name))} (power) (pin_numbers hide) (pin_names (offset 0) hide) (exclude_from_sim no) (in_bom yes) (on_board yes)
      (property "Reference" "#PWR" (at 0 ${gnd ? -6.35 : 3.81} 0) ${FONT_HIDE})
      (property "Value" ${q(name)} (at 0 ${gnd ? -3.81 : 2.286} 0) ${FONT})
      (property "Footprint" "" (at 0 0 0) ${FONT_HIDE})
      (property "Datasheet" "" (at 0 0 0) ${FONT_HIDE})
      (symbol ${q(pwrId(name) + '_0_1')}
      ${shape}
      )
      (symbol ${q(pwrId(name) + '_1_1')}
        (pin power_in line (at 0 0 ${gnd ? 270 : 90}) (length 0) hide (name ${q(name)} ${FONT}) (number "1" ${FONT}))
      )
    )`;
  }

  const PAPERS = [['A4', 297, 210], ['A3', 420, 297], ['A2', 594, 420], ['A1', 841, 594], ['A0', 1189, 841]];

  function sch(project, model) {
    const U = AK.util.uuid;
    const root = U();
    const proj = (project.name || 'autokicad').replace(/[^\w\-]+/g, '_');
    const b = model.bbox;
    const wmm = (b.x2 - b.x1) * MM, hmm = (b.y2 - b.y1) * MM;
    const paper = PAPERS.find(p => p[1] - 50 >= wmm && p[2] - 40 >= hmm) || PAPERS[PAPERS.length - 1];
    // сдвиг: центрируем на листе и выравниваем по сетке 1.27 мм
    // смещение — целое число U, иначе выводы уходят с сетки 1.27 мм
    const ox = Math.round(((paper[1] - wmm) / 2) / MM - b.x1);
    const oy = Math.round(((paper[2] - hmm) / 2) / MM - b.y1);
    const X = x => f((x + ox) * MM), Y = y => f((y + oy) * MM);

    const out = [];
    out.push(`(kicad_sch (version 20231120) (generator "autokicad") (generator_version "1.0")`);
    out.push(`  (uuid ${q(root)})`);
    out.push(`  (paper ${q(paper[0])})`);
    out.push(`  (title_block (title ${q(project.name || 'AutoKiCad')}) (comment 1 "Сгенерировано AutoKiCad из монтажной схемы"))`);

    out.push('  (lib_symbols');
    const used = [...new Set(model.symbols.map(s => s.def.type))];
    used.forEach(t => out.push('    ' + libSymbol(AK.LIB[t])));
    const pnames = [...new Set(model.power.map(p => p.name))];
    pnames.forEach(n => out.push('    ' + powerLib(n, AK.isGndName(n))));
    if (pnames.length) out.push('    ' + PWR_FLAG());
    out.push('  )');

    model.junctions.forEach(([x, y]) => out.push(`  (junction (at ${X(x)} ${Y(y)}) (diameter 0) (color 0 0 0 0) (uuid ${q(U())}))`));
    model.ncs.forEach(([x, y]) => out.push(`  (no_connect (at ${X(x)} ${Y(y)}) (uuid ${q(U())}))`));
    model.wires.forEach(w => out.push(`  (wire (pts (xy ${X(w.x1)} ${Y(w.y1)}) (xy ${X(w.x2)} ${Y(w.y2)})) (stroke (width 0) (type default)) (uuid ${q(U())}))`));
    const LANG = { R: 0, U: 90, L: 180, D: 270 };
    model.labels.forEach(l => {
      const just = (l.dir === 'L' || l.dir === 'D') ? 'right bottom' : 'left bottom';
      out.push(`  (label ${q(l.text)} (at ${X(l.x)} ${Y(l.y)} ${LANG[l.dir]}) (fields_autoplaced yes) (effects (font (size 1.27 1.27)) (justify ${just})) (uuid ${q(U())}))`);
    });

    const inst = ref => `(instances (project ${q(proj)} (path ${q('/' + root)} (reference ${q(ref)}) (unit 1))))`;

    model.symbols.forEach(sy => {
      const d = sy.def, s = d.sym, c = sy.comp;
      out.push(`  (symbol (lib_id ${q('AutoKiCad:' + d.kicadName)}) (at ${X(sy.x)} ${Y(sy.y)} 0) (unit 1) (exclude_from_sim no) (in_bom yes) (on_board yes) (dnp no) (uuid ${q(U())})`);
      const just = a => a === 'start' ? ' (justify left)' : a === 'end' ? ' (justify right)' : '';
      out.push(`    (property "Reference" ${q(c.ref)} (at ${X(sy.x + s.ref[0])} ${Y(sy.y + s.ref[1])} 0) (effects (font (size 1.27 1.27))${just(s.ref[2])}))`);
      out.push(`    (property "Value" ${q(c.value || d.value)} (at ${X(sy.x + s.val[0])} ${Y(sy.y + s.val[1])} 0) (effects (font (size 1.27 1.27))${just(s.val[2])}))`);
      out.push(`    (property "Footprint" ${q(d.footprint || '')} (at ${X(sy.x)} ${Y(sy.y)} 0) ${FONT_HIDE})`);
      out.push(`    (property "Datasheet" "~" (at ${X(sy.x)} ${Y(sy.y)} 0) ${FONT_HIDE})`);
      Object.keys(s.pins).forEach(p => out.push(`    (pin ${q(p)} (uuid ${q(U())}))`));
      out.push(`    ${inst(c.ref)}`);
      out.push('  )');
    });

    model.power.forEach((p, i) => {
      const ref = '#PWR0' + String(i + 1).padStart(2, '0');
      const gnd = p.kind === 'gnd';
      out.push(`  (symbol (lib_id ${q('AutoKiCad:' + pwrId(p.name))}) (at ${X(p.x)} ${Y(p.y)} 0) (unit 1) (exclude_from_sim no) (in_bom yes) (on_board yes) (dnp no) (uuid ${q(U())})`);
      out.push(`    (property "Reference" ${q(ref)} (at ${X(p.x)} ${Y(p.y + (gnd ? 2.5 : -1.5))} 0) ${FONT_HIDE})`);
      out.push(`    (property "Value" ${q(p.name)} (at ${X(p.x)} ${Y(p.y + (gnd ? 1.5 : -0.9))} 0) ${FONT})`);
      out.push(`    (property "Footprint" "" (at ${X(p.x)} ${Y(p.y)} 0) ${FONT_HIDE})`);
      out.push(`    (property "Datasheet" "" (at ${X(p.x)} ${Y(p.y)} 0) ${FONT_HIDE})`);
      out.push(`    (pin "1" (uuid ${q(U())}))`);
      out.push(`    ${inst(ref)}`);
      out.push('  )');
    });

    // PWR_FLAG — по одному на цепь питания, в точке первого её символа (иначе ERC: «power pin not driven»)
    const flagged = new Set();
    model.power.forEach(p => {
      if (flagged.has(p.name)) return;
      flagged.add(p.name);
      const ref = '#FLG0' + String(flagged.size).padStart(2, '0');
      const rot = p.kind === 'gnd' ? 0 : 180; // флаг в сторону, свободную от символа питания
      out.push(`  (symbol (lib_id "AutoKiCad:PWR_FLAG") (at ${X(p.x)} ${Y(p.y)} ${rot}) (unit 1) (exclude_from_sim no) (in_bom yes) (on_board yes) (dnp no) (uuid ${q(U())})`);
      out.push(`    (property "Reference" ${q(ref)} (at ${X(p.x)} ${Y(p.y)} 0) ${FONT_HIDE})`);
      out.push(`    (property "Value" "PWR_FLAG" (at ${X(p.x)} ${Y(p.y)} 0) ${FONT_HIDE})`);
      out.push(`    (property "Footprint" "" (at ${X(p.x)} ${Y(p.y)} 0) ${FONT_HIDE})`);
      out.push(`    (property "Datasheet" "~" (at ${X(p.x)} ${Y(p.y)} 0) ${FONT_HIDE})`);
      out.push(`    (pin "1" (uuid ${q(U())}))`);
      out.push(`    ${inst(ref)}`);
      out.push('  )');
    });

    out.push('  (sheet_instances (path "/" (page "1")))');
    out.push(')');
    return out.join('\n') + '\n';
  }

  function netlist(project, nl) {
    const comps = project.components.filter(c => !AK.LIB[c.type].virtual);
    const out = [];
    out.push('(export (version "E")');
    out.push(`  (design (source ${q((project.name || 'autokicad') + '.kicad_sch')}) (date ${q(new Date().toISOString())}) (tool "AutoKiCad"))`);
    out.push('  (components');
    comps.forEach(c => {
      const d = AK.LIB[c.type];
      out.push(`    (comp (ref ${q(c.ref)}) (value ${q(c.value || d.value)}) (footprint ${q(d.footprint || '')}) (libsource (lib "AutoKiCad") (part ${q(d.kicadName)}) (description ${q(d.name)})) (sheetpath (names "/") (tstamps "/")) (tstamps ${q(AK.util.uuid())}))`);
    });
    out.push('  )');
    out.push('  (nets');
    let code = 1;
    nl.drawn.forEach(n => {
      const nodes = n.real.map(p => {
        const c = comps.find(x => x.id === p.c);
        return `(node (ref ${q(c.ref)}) (pin ${q(p.p)}) (pinfunction ${q(AK.pinName(c.type, p.p))}))`;
      });
      out.push(`    (net (code ${q(code++)}) (name ${q(n.name)}) ${nodes.join(' ')})`);
    });
    out.push('  )');
    out.push(')');
    return out.join('\n') + '\n';
  }

  // Библиотека символов проекта: те же символы, что встроены в схему
  function symLib(model) {
    const out = ['(kicad_symbol_lib (version 20231120) (generator "autokicad") (generator_version "1.0")'];
    [...new Set(model.symbols.map(s => s.def.type))].forEach(t => out.push('  ' + libSymbol(AK.LIB[t], false)));
    const pnames = [...new Set(model.power.map(p => p.name))];
    pnames.forEach(n => out.push('  ' + powerLib(n, AK.isGndName(n), false)));
    if (pnames.length) out.push('  ' + PWR_FLAG(false));
    out.push(')');
    return out.join('\n') + '\n';
  }

  // Готовый проект KiCad: .kicad_pro + .kicad_sch + библиотека + sym-lib-table, упакованные в zip
  function projectZip(project, model, base) {
    const files = [
      [`${base}/${base}.kicad_pro`, JSON.stringify({ meta: { filename: base + '.kicad_pro', version: 1 } }, null, 2) + '\n'],
      [`${base}/${base}.kicad_sch`, sch(project, model)],
      [`${base}/AutoKiCad.kicad_sym`, symLib(model)],

    ];
    return AK.util.zip(files);
  }

})();

// Экспорт в формат исходника схемы EasyEDA Standard (JSON, docType 1).
// Единицы EasyEDA: 10 px = 100 mil = 1 U.
window.AK = window.AK || {};

// opts.native — для расширения: компоненты из библиотеки EasyEDA вставляются «родными» символами
// в их исходных координатах; расширение затем сдвигает их на место (moves) и добавляет остальное (rest).
AK.exportEasyEDA = function (project, model, opts) {
  opts = opts || {};
  let gid = 1;
  const id = () => 'gge' + (gid++);
  const S = 10, OX = 400, OY = 300;
  const b = model.bbox;
  // целочисленное смещение: EasyEDA округляет координаты выводов до целых px
  const bx = Math.floor(b.x1), by = Math.floor(b.y1);
  const X = x => Math.round((x - bx) * S * 100) / 100 + OX;
  const Y = y => Math.round((y - by) * S * 100) / 100 + OY;
  const R2 = v => Math.round(v * 100) / 100;
  const clean = s => String(s == null ? '' : s).replace(/[~`^#@$]/g, '_');
  const BODY = '#880000', PIN = '#880000';
  const shapes = [], nativeLibs = [], moves = [];
  const libPara = (d, c) => {
    const cp = Object.assign({}, d.ee.head.c_para, { pre: c.ref });
    if (c.value && c.value !== d.value) cp.Value = c.value;
    return Object.entries(cp).map(([k, v]) => clean(k) + '`' + clean(v)).join('`') + '`';
  };

  model.symbols.forEach(sy => {
    const d = sy.def, s = d.sym, c = sy.comp;
    if (opts.native && d.ee) {
      const gId = id(), h = d.ee.head;
      // подписи обозначения и номинала (в библиотечном символе их нет — EasyEDA иначе ставит «?» в начало координат)
      const [ox, oy] = d.ee.origin, nx = u => R2(ox + u * S), ny = u => R2(oy + u * S);
      const val = c.value || d.value;
      const texts = [
        `T~P~${nx(s.ref[0])}~${ny(s.ref[1])}~0~#000080~Arial~~~~~comment~${clean(c.ref)}~1~middle~${id()}~0~pinpart`,
        `T~N~${nx(s.val[0])}~${ny(s.val[1]) + 2}~0~#000080~Arial~~~~~comment~${clean(val)}~1~middle~${id()}~0~pinpart`,
      ];
      const own = d.ee.shapes.filter(x => !/^T~[PN]~/.test(x));
      nativeLibs.push(`LIB~${h.x}~${h.y}~${libPara(d, c)}~0~0~${gId}~${h.puuid || ''}~${h.uuid || ''}~0#@$${texts.concat(own).join('#@$')}`);
      const dx = X(sy.x) - d.ee.origin[0], dy = Y(sy.y) - d.ee.origin[1];
      moves.push({ id: gId, ref: c.ref, dx, dy });
      // вывод символа не на сетке 100 mil — короткий провод до точки, куда пришла трасса
      Object.values(s.pins).forEach(p => {
        if (!p.off) return;
        const ax0 = p.ee[0] + dx, ay0 = p.ee[1] + dy, bx0 = X(sy.x + p.x), by0 = Y(sy.y + p.y);
        shapes.push(`W~${ax0} ${ay0} ${bx0} ${ay0} ${bx0} ${by0}~#008800~1~0~none~${id()}~0`);
      });
      return;
    }
    const ax = u => X(sy.x + u), ay = u => Y(sy.y + u);
    const sub = [];
    sub.push(`T~P~${ax(s.ref[0])}~${ay(s.ref[1])}~0~#000080~Arial~~~~~comment~${clean(c.ref)}~1~${s.ref[2]}~${id()}~0~pinpart`);
    sub.push(`T~N~${ax(s.val[0])}~${ay(s.val[1]) + 2}~0~#000080~Arial~~~~~comment~${clean(c.value || d.value)}~1~${s.val[2]}~${id()}~0~pinpart`);
    s.gfx.forEach(g => {
      const sw = g.w ? g.w : 1;
      if (g.t === 'line') sub.push(`PL~${g.pts.map(p => ax(p[0]) + ' ' + ay(p[1])).join(' ')}~${BODY}~${sw}~0~none~${id()}~0`);
      else if (g.t === 'poly') sub.push(`PG~${g.pts.map(p => ax(p[0]) + ' ' + ay(p[1])).join(' ')}~${BODY}~1~0~${BODY}~${id()}~0`);
      else if (g.t === 'rect') sub.push(`R~${ax(g.x1)}~${ay(g.y1)}~~~${(g.x2 - g.x1) * S}~${(g.y2 - g.y1) * S}~${BODY}~1~0~${g.fill === 'body' ? '#FFFFCC' : 'none'}~${id()}~0`);
      else if (g.t === 'circle') sub.push(`E~${ax(g.cx)}~${ay(g.cy)}~${g.r * S}~${g.r * S}~${BODY}~1~0~none~${id()}~0`);
      else if (g.t === 'text') sub.push(`T~L~${ax(g.x)}~${ay(g.y) + 3}~0~${BODY}~Arial~7pt~~~~comment~${clean(g.text)}~1~middle~${id()}~0~pinpart`);
    });
    const ROT = { R: 180, L: 0, U: 90, D: 270 };
    Object.entries(s.pins).forEach(([num, p]) => {
      const px = ax(p.x), py = ay(p.y);
      const [dx, dy] = AK.DIRV[p.dir];
      const len = p.len * S;
      const path = dx ? `M ${px} ${py} h ${dx * len}` : `M ${px} ${py} v ${dy * len}`;
      const ex = px + dx * len, ey = py + dy * len;
      const name = clean(p.name || num);
      const nameVis = s.showNames && p.name ? 1 : 0;
      const numVis = s.showNums ? 1 : 0;
      // у вертикального вывода имя — по центру, внутри корпуса
      const anchor = dy ? 'middle' : p.dir === 'R' ? 'start' : 'end';
      sub.push(`P~show~0~${clean(num)}~${px}~${py}~${ROT[p.dir]}~${id()}~0^^${px}~${py}^^${path}~${PIN}^^` +
        `${nameVis}~${ex + dx * 3}~${ey + 3 + dy * 5}~0~${name}~${anchor}~~~#0000FF^^` +
        `${numVis}~${(px + ex) / 2}~${py - 1}~0~${clean(num)}~middle~~~#0000FF^^0~${ex}~${ey}^^0~M 0 0`);
    });
    const para = d.ee ? libPara(d, c) : [
      'package', clean(d.eePackage || ''), 'pre', clean(c.ref), 'spicePre', d.prefix,
      'nameAlias', 'Value', 'Value', clean(c.value || d.value), 'Footprint(KiCad)', clean(d.footprint || ''), 'Contributor', 'AutoKiCad',
    ].join('`') + '`';
    // поля заголовка: x, y, c_para, rotation, importFlag, gId, puuid, uuid, locked
    shapes.push(`LIB~${X(sy.x)}~${Y(sy.y)}~${para}~0~0~${id()}~~~0#@$${sub.join('#@$')}`);
  });

  model.wires.forEach(w => shapes.push(`W~${X(w.x1)} ${Y(w.y1)} ${X(w.x2)} ${Y(w.y2)}~#008800~1~0~none~${id()}~0`));
  model.junctions.forEach(([x, y]) => shapes.push(`J~${X(x)}~${Y(y)}~2.5~#CC0000~${id()}~0`));
  model.ncs.forEach(([x, y]) => {
    const px = X(x), py = Y(y);
    shapes.push(`O~${px}~${py}~${id()}~M ${px - 4} ${py - 4} L ${px + 4} ${py + 4} M ${px + 4} ${py - 4} L ${px - 4} ${py + 4}~#33cc33~0`);
  });
  // метка влево — без поворота, с выравниванием по правому краю (поворот 180° переворачивает текст)
  const LROT = { R: 0, U: 270, L: 0, D: 90 };
  model.labels.forEach(l => {
    const px = X(l.x), py = Y(l.y);
    const anchor = (l.dir === 'L' || l.dir === 'D') ? 'end' : 'start';
    shapes.push(`N~${px}~${py}~${LROT[l.dir]}~#0000FF~${clean(l.text)}~${id()}~${anchor}~${px + (anchor === 'start' ? 2 : -2)}~${py - 3}~Times New Roman~~0`);
  });
  model.power.forEach(p => {
    const px = X(p.x), py = Y(p.y);
    if (p.kind === 'gnd') {
      shapes.push(`F~part_netLabel_gnD~${px}~${py}~0~${id()}~~0^^${px}~${py}^^${clean(p.name)}~#000000~${px - 10}~${py + 25}~0~start~${p.name === 'GND' ? 0 : 1}~Times New Roman~9pt^^` +
        `PL~${px} ${py + 5} ${px} ${py}~#000000~1~0~transparent~${id()}~0^^PL~${px - 6} ${py + 5} ${px + 6} ${py + 5}~#000000~1~0~transparent~${id()}~0^^` +
        `PL~${px - 4} ${py + 7.5} ${px + 4} ${py + 7.5}~#000000~1~0~transparent~${id()}~0^^PL~${px - 1.5} ${py + 10} ${px + 1.5} ${py + 10}~#000000~1~0~transparent~${id()}~0`);
    } else {
      shapes.push(`F~part_netLabel_VCC~${px}~${py}~180~${id()}~~0^^${px}~${py}^^${clean(p.name)}~#000000~${px - 8}~${py - 8}~0~start~1~Times New Roman~9pt^^` +
        `PL~${px} ${py} ${px} ${py - 6}~#000000~1~0~transparent~${id()}~0^^PL~${px - 5} ${py - 6} ${px + 5} ${py - 6}~#000000~1~0~transparent~${id()}~0`);
    }
  });

  const w = (b.x2 - b.x1) * S, h = (b.y2 - b.y1) * S;
  if (opts.native) {
    const head = { docType: '1', editorVersion: '6.5.51', newgId: true, c_para: { 'Prefix Start': '1' }, c_spiceCmd: 'null', hasIdFlag: true, x: '0', y: '0', importFlag: 0, transformList: '' };
    const libDoc = {
      head, canvas: `CA~${Math.max(1200, w + 800)}~${Math.max(1000, h + 600)}~#FFFFFF~yes~#CCCCCC~5~1000~1000~line~5~pixel~5~0~0`,
      shape: nativeLibs, BBox: { x: OX - 20, y: OY - 20, width: w + 40, height: h + 40 }, colors: {},
    };
    return { libDoc, moves, rest: shapes };
  }
  const doc = {
    head: {
      docType: '1', editorVersion: '6.5.40', newgId: true, c_para: { 'Prefix Start': '1' }, c_spiceCmd: 'null',
      hasIdFlag: true, x: '0', y: '0', importFlag: 0, transformList: '',
    },
    canvas: `CA~${Math.max(1200, w + 800)}~${Math.max(1000, h + 600)}~#FFFFFF~yes~#CCCCCC~5~1000~1000~line~5~pixel~5~0~0`,
    shape: shapes,
    BBox: { x: OX - 20, y: OY - 20, width: w + 40, height: h + 40 },
    colors: {},
  };
  return JSON.stringify(doc, null, 1);
};

// ---- js/editor.js ----
// Редактор «Монтаж»: компоненты в реальном виде и провода между выводами
window.AK = window.AK || {};

AK.WIRE_COLORS = ['#e53935', '#212121', '#1e88e5', '#43a047', '#fdd835', '#fb8c00', '#8e24aa', '#eceff1'];

AK.Editor = class {
  constructor(svg, app) {
    this.svg = svg; this.app = app;
    this.view = { x: 400, y: 300, z: 1.8 };
    this.wireColor = 3; // сигнальные перемычки — зелёные
    svg.innerHTML = `<defs><pattern id="bgrid" width="10" height="10" patternUnits="userSpaceOnUse"><circle cx="0" cy="0" r=".9" fill="#c9d1db"/></pattern></defs>
      <g class="root">
        <rect x="-10000" y="-10000" width="20000" height="20000" fill="url(#bgrid)" class="bg"/>
        <g class="l-comps"></g><g class="l-wires"></g><g class="l-labels"></g><g class="l-pins"></g><g class="l-temp"></g>
      </g>`;
    this.root = svg.querySelector('.root');
    this.L = {
      comps: svg.querySelector('.l-comps'), wires: svg.querySelector('.l-wires'), labels: svg.querySelector('.l-labels'),
      pins: svg.querySelector('.l-pins'), temp: svg.querySelector('.l-temp'),
    };
    this.bind();
    this.applyView();
  }

  get P() { return this.app.project; }
  comp(id) { return this.P.components.find(c => c.id === id); }

  toWorld(e) {
    const r = this.svg.getBoundingClientRect();
    return [(e.clientX - r.left - this.view.x) / this.view.z, (e.clientY - r.top - this.view.y) / this.view.z];
  }
  applyView() {
    this.root.setAttribute('transform', `translate(${this.view.x} ${this.view.y}) scale(${this.view.z})`);
  }
  zoomBy(f, cx, cy) {
    const r = this.svg.getBoundingClientRect();
    if (cx == null) { cx = r.width / 2; cy = r.height / 2; }
    const nz = Math.max(0.3, Math.min(8, this.view.z * f));
    this.view.x = cx - (cx - this.view.x) * nz / this.view.z;
    this.view.y = cy - (cy - this.view.y) * nz / this.view.z;
    this.view.z = nz;
    this.applyView();
  }
  fit() {
    const r = this.svg.getBoundingClientRect();
    if (!this.P.components.length || !r.width) return;
    let x1 = Infinity, y1 = Infinity, x2 = -Infinity, y2 = -Infinity;
    this.P.components.forEach(c => { const b = AK.compWorldBox(c); x1 = Math.min(x1, b.x); y1 = Math.min(y1, b.y); x2 = Math.max(x2, b.x + b.w); y2 = Math.max(y2, b.y + b.h); });
    this.P.wires.forEach(w => w.points.forEach(([x, y]) => { x1 = Math.min(x1, x); y1 = Math.min(y1, y); x2 = Math.max(x2, x); y2 = Math.max(y2, y); }));
    const z = Math.min(r.width / (x2 - x1 + 120), r.height / (y2 - y1 + 120), 4);
    this.view = { z, x: r.width / 2 - (x1 + x2) / 2 * z, y: r.height / 2 - (y1 + y2) / 2 * z };
    this.applyView();
  }

  wirePts(w) {
    const a = this.comp(w.from.c), b = this.comp(w.to.c);
    return [AK.pinWorld(a, w.from.p), ...w.points, AK.pinWorld(b, w.to.p)];
  }

  // ---------------- отрисовка ----------------
  render() {
    const P = this.P, sel = this.app.sel, e = AK.util.esc;
    // занятые выводы: к ним подходит провод или в той же точке стоит другой вывод (ножка в отверстии)
    const connected = new Set(), at = new Map();
    P.wires.forEach(w => { connected.add(w.from.c + ':' + w.from.p); connected.add(w.to.c + ':' + w.to.p); });
    P.components.forEach(c => Object.keys(AK.LIB[c.type].visual.pins).forEach(p => {
      const s = AK.pinWorld(c, p).map(Math.round).join(',');
      if (!at.has(s)) at.set(s, []);
      at.get(s).push(c.id + ':' + p);
    }));
    const isLeg = k => !AK.LIB[this.comp(k.split(':')[0]).type].board;
    at.forEach(ks => { if (ks.length > 1 && ks.some(isLeg)) ks.forEach(k => connected.add(k)); });

    let comps = '', labels = '', pins = '';
    const selSet = new Set(this.selIds());
    // макетные платы — под остальными компонентами
    const order = P.components.filter(c => AK.LIB[c.type].board).concat(P.components.filter(c => !AK.LIB[c.type].board));
    order.forEach(c => {
      const d = AK.LIB[c.type];
      const [bx, by, bw, bh] = d.visual.hit || d.visual.box;
      const isSel = selSet.has(c.id);
      comps += `<g class="comp${d.board ? ' board' : ''}${isSel ? ' sel' : ''}" data-id="${c.id}" transform="translate(${c.x} ${c.y}) rotate(${c.rot || 0})">` +
        d.visual.draw(c, { on: connected }) + (d.board ? '' : `<rect class="hit" x="${bx}" y="${by}" width="${bw}" height="${bh}" fill="transparent"/>`) + '</g>';
      const wb = AK.compWorldBox(c);
      if (isSel) labels += `<rect class="selbox" x="${wb.x - 4}" y="${wb.y - 4}" width="${wb.w + 8}" height="${wb.h + 8}" rx="4"/>`;
      const label = e(c.ref) + (c.value && d.type !== 'nano' && d.type !== 'ne555' ? ' · ' + e(c.value) : '');
      // у вертикально стоящих компонентов подпись сбоку, чтобы не закрывать соседние отверстия
      if (!d.virtual && wb.h > wb.w * 1.6 && d.type !== 'electrolytic') labels += `<text class="ref side" x="${wb.x + wb.w + 3}" y="${wb.y + wb.h / 2 + 3}">${label}</text>`;
      else if (!d.virtual) labels += `<text class="ref" x="${wb.x + wb.w / 2}" y="${wb.y - 6}">${label}</text>`;
      if (d.board) return; // отверстия платы нарисованы в её корпусе
      Object.keys(d.visual.pins).forEach(p => {
        const [x, y] = AK.pinWorld(c, p);
        const k = c.id + ':' + p;
        pins += `<circle class="pin${connected.has(k) ? ' on' : ''}" data-k="${k}" cx="${x}" cy="${y}" r="3.2"/>`;
      });
    });

    let wires = '';
    P.wires.forEach(w => {
      const pts = this.wirePts(w).map(p => p.join(',')).join(' ');
      const isSel = sel && sel.kind === 'wire' && sel.id === w.id;
      const col = w.color || AK.WIRE_COLORS[0];
      wires += `<g class="wire${isSel ? ' sel' : ''}" data-id="${w.id}" data-k1="${w.from.c}:${w.from.p}" data-k2="${w.to.c}:${w.to.p}">` +
        `<polyline class="w-hit" points="${pts}"/><polyline class="w-shadow" points="${pts}"/>` +
        `<polyline class="w-main" points="${pts}" stroke="${col}"/>`;
      if (isSel) w.points.forEach(([x, y], i) => { wires += `<circle class="handle" data-i="${i}" cx="${x}" cy="${y}" r="3.5"/>`; });
      if (w.net) {
        const all = this.wirePts(w), m = all[Math.floor((all.length - 1) / 2)], n2 = all[Math.floor((all.length - 1) / 2) + 1];
        wires += `<text class="wname" x="${(m[0] + n2[0]) / 2}" y="${(m[1] + n2[1]) / 2 - 5}">${e(w.net)}</text>`;
      }
      wires += '</g>';
    });

    this.L.comps.innerHTML = comps;
    this.L.wires.innerHTML = wires;
    this.L.labels.innerHTML = labels;
    this.L.pins.innerHTML = pins;
    this.renderTemp();
  }

  renderTemp() {
    let s = '';
    if (this.wiring) {
      const a = AK.pinWorld(this.comp(this.wiring.from.c), this.wiring.from.p);
      const pts = [a, ...this.wiring.points, this.wiring.cur || a].map(p => p.join(',')).join(' ');
      s += `<polyline class="w-temp" points="${pts}" stroke="${AK.WIRE_COLORS[this.wireColor]}"/>`;
    }
    if (this.box) {
      const b = this.box, x = Math.min(b.x0, b.x1), y = Math.min(b.y0, b.y1);
      s += `<rect class="selrect" x="${x}" y="${y}" width="${Math.abs(b.x1 - b.x0)}" height="${Math.abs(b.y1 - b.y0)}"/>`;
    }
    if (this.placing && this.placing.cur) {
      const d = AK.LIB[this.placing.type];
      const fake = { type: this.placing.type, value: d.value, ref: d.prefix + '?', props: this.defaultProps(d), rot: 0 };
      s += `<g class="ghost" transform="translate(${this.placing.cur[0]} ${this.placing.cur[1]})">${d.visual.draw(fake)}</g>`;
    }
    this.L.temp.innerHTML = s;
  }

  defaultProps(d) {
    const p = {};
    (d.props || []).forEach(x => { p[x.key] = x.def; });
    return p;
  }

  // ---------------- операции ----------------
  addComponent(type, x, y) {
    const d = AK.LIB[type];
    const [sx, sy] = AK.snapPos(type, 0, x, y);
    const c = {
      id: AK.util.uid('c'), type, ref: this.app.nextRef(d.prefix, d.virtual), value: d.value,
      x: sx, y: sy, rot: 0, props: this.defaultProps(d),
    };
    if (d.props && d.props.some(p => p.key === 'net')) c.value = c.props.net;
    this.P.components.push(c);
    return c;
  }

  // ---------------- выделение ----------------
  // выделенные компоненты: один (kind 'comp') или группа (kind 'multi')
  selIds() {
    const s = this.app.sel;
    if (!s) return [];
    if (s.kind === 'comp') return [s.id];
    if (s.kind === 'multi') return s.ids.filter(id => this.comp(id));
    return [];
  }
  setSel(ids) {
    this.app.select(!ids.length ? null : ids.length === 1 ? { kind: 'comp', id: ids[0] } : { kind: 'multi', ids });
  }
  selectAll() {
    this.cancel();
    this.setSel(this.P.components.map(c => c.id));
    this.render();
  }

  // что двигается вместе с выделением: сами компоненты, вставленные в выделенные платы,
  // и провода, у которых оба конца в этом наборе (их точки изгиба)
  moveSet(ids) {
    const set = new Set(ids);
    const pk = o => Object.keys(AK.LIB[o.type].visual.pins).map(p => AK.pinWorld(o, p).map(Math.round).join(','));
    ids.forEach(id => {
      const b = this.comp(id);
      if (!b || !AK.LIB[b.type].board) return;
      const holes = new Set(pk(b));
      this.P.components.forEach(o => { if (!AK.LIB[o.type].board && pk(o).some(s => holes.has(s))) set.add(o.id); });
    });
    return {
      comps: this.P.components.filter(c => set.has(c.id)),
      wires: this.P.wires.filter(w => set.has(w.from.c) && set.has(w.to.c)),
    };
  }
  shiftSet(ms, dx, dy) {
    ms.comps.forEach(o => { o.x += dx; o.y += dy; });
    ms.wires.forEach(w => { w.points = w.points.map(([x, y]) => [x + dx, y + dy]); });
  }

  nudge(dx, dy) {
    const ids = this.selIds();
    if (!ids.length) return;
    this.app.checkpoint();
    this.shiftSet(this.moveSet(ids), dx, dy);
    this.app.changed({ keepView: true });
  }

  deleteSelection() {
    const sel = this.app.sel;
    if (!sel) return;
    this.app.checkpoint();
    const ids = new Set(this.selIds());
    if (ids.size) {
      this.P.components = this.P.components.filter(c => !ids.has(c.id));
      this.P.wires = this.P.wires.filter(w => !ids.has(w.from.c) && !ids.has(w.to.c));
    } else {
      this.P.wires = this.P.wires.filter(w => w.id !== sel.id);
    }
    this.app.select(null);
    this.app.changed();
  }

  rotateSelection() {
    const ids = this.selIds();
    if (!ids.length) return;
    this.app.checkpoint();
    const ms = this.moveSet(ids);
    if (ms.comps.length === 1) {
      const c = ms.comps[0];
      c.rot = ((c.rot || 0) + 90) % 360;
      [c.x, c.y] = AK.snapPos(c.type, c.rot, c.x, c.y);
    } else {
      // группа поворачивается вокруг узла сетки в центре — выводы остаются в отверстиях
      let x1 = Infinity, y1 = Infinity, x2 = -Infinity, y2 = -Infinity;
      ms.comps.forEach(c => { const b = AK.compWorldBox(c); x1 = Math.min(x1, b.x); y1 = Math.min(y1, b.y); x2 = Math.max(x2, b.x + b.w); y2 = Math.max(y2, b.y + b.h); });
      const gx = Math.round((x1 + x2) / 20) * 10, gy = Math.round((y1 + y2) / 20) * 10;
      const R = ([x, y]) => { const [rx, ry] = AK.util.rotPt(x - gx, y - gy, 90); return [gx + rx, gy + ry]; };
      ms.comps.forEach(c => { [c.x, c.y] = R([c.x, c.y]); c.rot = ((c.rot || 0) + 90) % 360; });
      ms.wires.forEach(w => { w.points = w.points.map(R); });
    }
    this.app.changed();
  }

  // ---------------- буфер обмена ----------------
  makeClip(ids) {
    const ms = this.moveSet(ids);
    if (!ms.comps.length) return null;
    const ee = {};
    ms.comps.forEach(c => { const d = AK.LIB[c.type]; if (d.ee && this.P.eeParts && this.P.eeParts[d.ee.uuid]) ee[d.ee.uuid] = this.P.eeParts[d.ee.uuid]; });
    // ширина группы — копия ставится справа от оригинала, не налезая на него (иначе отверстия плат совпадут)
    let x1 = Infinity, x2 = -Infinity;
    ms.comps.forEach(c => { const b = AK.compWorldBox(c); x1 = Math.min(x1, b.x); x2 = Math.max(x2, b.x + b.w); });
    ms.wires.forEach(w => w.points.forEach(([x]) => { x1 = Math.min(x1, x); x2 = Math.max(x2, x); }));
    const step = Math.ceil((x2 - x1 + 20) / 10) * 10;
    return JSON.parse(JSON.stringify({ comps: ms.comps, wires: ms.wires, eeParts: ee, step }));
  }
  copySelection() {
    const clip = this.makeClip(this.selIds());
    if (!clip) return false;
    AK.clipboard = clip;
    this.pasteN = 0;
    try { localStorage.setItem('autokicad.clipboard', JSON.stringify(clip)); } catch (e) { /* хранилище недоступно */ }
    return true;
  }
  paste() {
    let clip = AK.clipboard;
    if (!clip) { try { clip = JSON.parse(localStorage.getItem('autokicad.clipboard')); } catch (e) { clip = null; } }
    if (!clip || !clip.comps || !clip.comps.length) { this.app.toast('Буфер пуст: выделите компоненты и нажмите Ctrl+C'); return; }
    this.pasteN = (this.pasteN || 0) + 1;
    this.insertClip(clip, clip.step * this.pasteN, 0);
  }
  duplicateSelection() {
    const clip = this.makeClip(this.selIds());
    if (clip) this.insertClip(clip, clip.step, 0);
  }
  insertClip(clip, dx, dy) {
    this.cancel();
    this.app.checkpoint();
    if (Object.keys(clip.eeParts || {}).length) {
      this.P.eeParts = Object.assign({}, clip.eeParts, this.P.eeParts);
      AK.EE.register(this.P);
    }
    const map = {}, ids = [];
    clip.comps.forEach(src => {
      const d = AK.LIB[src.type];
      if (!d) return;
      const c = JSON.parse(JSON.stringify(src));
      c.id = AK.util.uid('c');
      c.ref = this.app.nextRef(d.prefix, d.virtual);
      c.x += dx; c.y += dy;
      map[src.id] = c.id;
      this.P.components.push(c);
      ids.push(c.id);
    });
    clip.wires.forEach(src => {
      if (!map[src.from.c] || !map[src.to.c]) return;
      this.P.wires.push(Object.assign(JSON.parse(JSON.stringify(src)), {
        id: AK.util.uid('w'), from: { c: map[src.from.c], p: src.from.p }, to: { c: map[src.to.c], p: src.to.p },
        points: src.points.map(([x, y]) => [x + dx, y + dy]),
      }));
    });
    this.setSel(ids);
    this.app.changed();
  }

  cancel() {
    if (this.wiring || this.placing) {
      this.wiring = null; this.placing = null;
      this.renderTemp();
      this.app.hint('');
      return true;
    }
    return false;
  }

  startPlacing(type) {
    this.cancel();
    this.placing = { type };
    this.app.hint(`Кликните на поле, чтобы поставить «${AK.LIB[type].name}». Esc — отмена.`);
  }

  finishWire(k) {
    const [c, p] = k.split(':');
    const f = this.wiring.from;
    if (c === f.c && p === f.p) return false;
    this.app.checkpoint();
    // провод к шине «−» / земле — чёрный, к шине «+» — красный, остальные — выбранного цвета
    const ends = [f, { c, p }].map(x => { const o = this.comp(x.c); return AK.LIB[o.type].board ? AK.pinName(o.type, x.p) : o.type; });
    let color = AK.WIRE_COLORS[this.wireColor];
    if (ends.some(s => /−|^gnd$/.test(s))) color = AK.WIRE_COLORS[1];
    else if (ends.some(s => /\+|^power$/.test(s))) color = AK.WIRE_COLORS[0];
    const w = { id: AK.util.uid('w'), from: f, to: { c, p }, points: this.wiring.points, color, net: '' };
    this.P.wires.push(w);
    this.wiring = null;
    this.app.hint('');
    this.app.select({ kind: 'wire', id: w.id });
    this.app.changed();
    return true;
  }

  // ---------------- события ----------------
  bind() {
    const svg = this.svg;
    const snap5 = v => Math.round(v / 5) * 5;

    svg.addEventListener('wheel', e => {
      e.preventDefault();
      const r = svg.getBoundingClientRect();
      this.zoomBy(e.deltaY < 0 ? 1.15 : 1 / 1.15, e.clientX - r.left, e.clientY - r.top);
    }, { passive: false });

    svg.addEventListener('contextmenu', e => { e.preventDefault(); if (!this.panMoved) this.cancel(); });

    svg.addEventListener('pointerdown', e => {
      const [wx, wy] = this.toWorld(e);
      // сдвиг поля — средней или правой кнопкой (как в EasyEDA), либо Alt + левой
      if (e.button === 1 || e.button === 2 || (e.button === 0 && e.altKey)) {
        this.panMoved = false;
        this.pan = { x: e.clientX, y: e.clientY, vx: this.view.x, vy: this.view.y };
        svg.setPointerCapture(e.pointerId);
        return;
      }
      if (e.button !== 0) return;

      if (this.placing) {
        this.app.checkpoint();
        const c = this.addComponent(this.placing.type, wx, wy);
        if (!e.shiftKey) { this.placing = null; this.app.hint(''); }
        this.app.select({ kind: 'comp', id: c.id });
        this.app.changed();
        return;
      }

      const pinEl = e.target.closest('.pin');
      if (this.wiring) {
        if (pinEl) {
          this.finishWire(pinEl.dataset.k);
        } else {
          this.wiring.points.push([snap5(wx), snap5(wy)]);
          this.renderTemp();
        }
        return;
      }

      if (pinEl) {
        const [c, p] = pinEl.dataset.k.split(':');
        // провод можно протянуть мышью (отпустить над другим выводом) или поставить двумя кликами
        this.wiring = { from: { c, p }, points: [], cur: [wx, wy], down: true };
        this.app.hint('Отпустите или кликните над другим выводом / отверстием. Клик по пустому месту — точка изгиба. Esc / ПКМ — отмена.');
        this.renderTemp();
        return;
      }

      const handle = e.target.closest('.handle');
      if (handle) {
        const wid = handle.closest('.wire').dataset.id;
        this.hdrag = { wid, i: +handle.dataset.i, moved: false };
        svg.setPointerCapture(e.pointerId);
        return;
      }

      const wireEl = e.target.closest('.wire');
      if (wireEl) {
        // без перерисовки при повторном клике, иначе dblclick теряет цель
        const s = this.app.sel;
        if (!(s && s.kind === 'wire' && s.id === wireEl.dataset.id)) {
          this.app.select({ kind: 'wire', id: wireEl.dataset.id });
          this.render();
        }
        return;
      }

      const compEl = e.target.closest('.comp');
      const add = e.shiftKey || e.ctrlKey || e.metaKey;
      const onBoard = compEl && AK.LIB[this.comp(compEl.dataset.id).type].board;
      if (compEl && !(add && onBoard)) {
        const c = this.comp(compEl.dataset.id);
        let ids = this.selIds();
        if (add) {
          // Ctrl/Shift + клик — добавить компонент в выделение или убрать из него
          ids = ids.includes(c.id) ? ids.filter(id => id !== c.id) : ids.concat(c.id);
          this.setSel(ids);
          this.render();
          return;
        }
        if (!ids.includes(c.id)) { ids = [c.id]; this.setSel(ids); }
        // вместе с компонентом едет всё выделенное; плата везёт вставленные в неё компоненты и перемычки
        this.drag = { id: c.id, ox: wx - c.x, oy: wy - c.y, moved: false, set: this.moveSet(ids) };
        svg.setPointerCapture(e.pointerId);
        this.render();
        return;
      }

      // рамка выделения: по пустому месту, или с Shift/Ctrl — и поверх платы
      this.box = { x0: wx, y0: wy, x1: wx, y1: wy, add, base: add ? this.selIds() : [], sx: e.clientX, sy: e.clientY };
      svg.setPointerCapture(e.pointerId);
    });

    svg.addEventListener('pointermove', e => {
      const [wx, wy] = this.toWorld(e);
      if (this.pan) {
        if (Math.abs(e.clientX - this.pan.x) + Math.abs(e.clientY - this.pan.y) > 3) this.panMoved = true;
        this.view.x = this.pan.vx + e.clientX - this.pan.x;
        this.view.y = this.pan.vy + e.clientY - this.pan.y;
        this.applyView();
        return;
      }
      if (this.box) {
        this.box.x1 = wx; this.box.y1 = wy;
        this.renderTemp();
        return;
      }
      if (this.drag) {
        const c = this.comp(this.drag.id);
        const [nx, ny] = AK.snapPos(c.type, c.rot, wx - this.drag.ox, wy - this.drag.oy);
        if (nx !== c.x || ny !== c.y) {
          if (!this.drag.moved) { this.app.checkpoint(); this.drag.moved = true; }
          this.shiftSet(this.drag.set, nx - c.x, ny - c.y);
          this.render();
        }
        return;
      }
      if (this.hdrag) {
        const w = this.P.wires.find(x => x.id === this.hdrag.wid);
        if (!this.hdrag.moved) { this.app.checkpoint(); this.hdrag.moved = true; }
        w.points[this.hdrag.i] = [snap5(wx), snap5(wy)];
        this.render();
        return;
      }
      if (this.wiring) { this.wiring.cur = [wx, wy]; this.renderTemp(); }
      if (this.placing) { this.placing.cur = AK.snapPos(this.placing.type, 0, wx, wy); this.renderTemp(); }
      this.hover(e);
    });

    const up = e => {
      if (this.wiring && this.wiring.down) {
        this.wiring.down = false;
        const pinEl = e.target && e.target.closest && e.target.closest('.pin');
        if (pinEl && this.finishWire(pinEl.dataset.k)) return;
      }
      if (this.box) {
        const b = this.box;
        this.box = null;
        if (Math.abs(e.clientX - b.sx) + Math.abs(e.clientY - b.sy) < 4) {
          // простой клик по пустому месту — снять выделение
          if (!b.add) this.app.select(null);
        } else {
          // выделяются компоненты, целиком попавшие в рамку
          const x1 = Math.min(b.x0, b.x1), x2 = Math.max(b.x0, b.x1), y1 = Math.min(b.y0, b.y1), y2 = Math.max(b.y0, b.y1);
          const inside = this.P.components.filter(c => {
            const r = AK.compWorldBox(c);
            return r.x >= x1 && r.y >= y1 && r.x + r.w <= x2 && r.y + r.h <= y2;
          }).map(c => c.id);
          this.setSel([...new Set(b.base.concat(inside))]);
        }
        this.render();
        return;
      }
      const moved = (this.drag && this.drag.moved) || (this.hdrag && this.hdrag.moved);
      this.drag = null; this.hdrag = null; this.pan = null;
      if (moved) this.app.changed();
    };
    svg.addEventListener('pointerup', up);
    svg.addEventListener('pointercancel', up);
    svg.addEventListener('pointerleave', () => this.app.tooltip(null));

    // двойной клик по проводу — добавить точку изгиба
    svg.addEventListener('dblclick', e => {
      const wireEl = e.target.closest('.wire');
      if (!wireEl) return;
      const w = this.P.wires.find(x => x.id === wireEl.dataset.id);
      const [wx, wy] = this.toWorld(e);
      const pts = this.wirePts(w);
      let best = 0, bd = Infinity;
      for (let i = 0; i < pts.length - 1; i++) {
        const d = distSeg(wx, wy, pts[i], pts[i + 1]);
        if (d < bd) { bd = d; best = i; }
      }
      this.app.checkpoint();
      w.points.splice(best, 0, [snap5(wx), snap5(wy)]);
      this.app.select({ kind: 'wire', id: w.id });
      this.app.changed();
    });

    // перетаскивание из палитры
    svg.addEventListener('dragover', e => { e.preventDefault(); e.dataTransfer.dropEffect = 'copy'; });
    svg.addEventListener('drop', e => {
      e.preventDefault();
      const type = e.dataTransfer.getData('text/x-akc-type');
      if (!type || !AK.LIB[type]) return;
      const [wx, wy] = this.toWorld(e);
      this.app.checkpoint();
      const c = this.addComponent(type, wx, wy);
      this.app.select({ kind: 'comp', id: c.id });
      this.app.changed();
    });

    function distSeg(x, y, a, b) {
      const dx = b[0] - a[0], dy = b[1] - a[1];
      const t = Math.max(0, Math.min(1, ((x - a[0]) * dx + (y - a[1]) * dy) / (dx * dx + dy * dy || 1)));
      return Math.hypot(x - a[0] - t * dx, y - a[1] - t * dy);
    }
  }

  hover(e) {
    const pinEl = e.target.closest && e.target.closest('.pin');
    const k = pinEl ? pinEl.dataset.k : null;
    if (k === this.hoverKey) { if (k) this.app.tooltip(this.tipText, e); return; }
    this.hoverKey = k;
    this.svg.querySelectorAll('.hl').forEach(n => n.classList.remove('hl'));
    if (!k) { this.app.tooltip(null); return; }
    const [cid, p] = k.split(':');
    const c = this.comp(cid);
    const nl = this.app.netlist();
    const net = nl.pinNet.get(k);
    const keys = new Set(net ? net.pins.map(x => x.c + ':' + x.p) : [k]);
    this.svg.querySelectorAll('.pin').forEach(n => { if (keys.has(n.dataset.k)) n.classList.add('hl'); });
    this.L.wires.querySelectorAll('.wire').forEach(n => { if (keys.has(n.dataset.k1)) n.classList.add('hl'); });
    const d = AK.LIB[c.type];
    this.tipText = `<b>${AK.util.esc(d.virtual ? d.name : c.ref)}</b> · вывод ${AK.util.esc(AK.pinName(c.type, p))}` +
      (net && net.name && net.drawn ? `<br>цепь <b>${AK.util.esc(net.name)}</b> (${net.real.length} выв.)` : '<br><i>не подключён</i>');
    this.app.tooltip(this.tipText, e);
  }
};

</script>

  buildPalette() {

      <div class="ee-attr">Источник: JLCEDA/EasyEDA Official Library · <a href="https://lceda.cn/" target="_blank" rel="noopener">lceda.cn</a> · <a href="https://easyeda.com" target="_blank" rel="noopener">easyeda.com</a></div>
      <div id="palProject"></div>`;
    Object.entries(cats).forEach(([cat, list]) => {
      html += `<div class="pal-cat" data-cat="${AK.util.esc(cat)}">${AK.util.esc(cat)}</div><div class="pal-grid" data-cat="${AK.util.esc(cat)}">`;
      list.forEach(d => { html += this.palItem(d); });

    });

  // фильтр встроенных компонентов по названию, номиналу и корпусу
  filterPalette(q) {
    const words = q.trim().toLowerCase().split(/\s+/).filter(Boolean);
    const pal = document.getElementById('palette');
    pal.querySelectorAll('.pal-grid[data-cat]').forEach(g => {
      let any = false;
      g.querySelectorAll('.pal-item').forEach(el => {
        const d = AK.LIB[el.dataset.type];
        const hay = [d.name, d.value, d.prefix, d.eePackage, d.kicadName, d.cat].join(' ').toLowerCase();
        el.hidden = !words.every(w => hay.includes(w));
        any = any || !el.hidden;
      });
      g.hidden = !any;
      g.previousElementSibling.hidden = !any;
    });
  }

  palItem(d) {
    const [bx, by, bw, bh] = d.visual.box;
    const pad = 6;
    const fake = { type: d.type, value: d.value, ref: d.prefix, rot: 0, props: this.editor.defaultProps(d) };
    return `<button class="pal-item" draggable="true" data-type="${d.type}" title="${AK.util.esc(d.name)}">` +
      `<svg viewBox="${bx - pad} ${by - pad} ${bw + pad * 2} ${bh + pad * 2}" preserveAspectRatio="xMidYMid meet">${d.visual.draw(fake)}</svg>` +
      `<span>${AK.util.esc(d.name)}</span></button>`;
  }
  bindPalItems(root) {
    root.querySelectorAll('.pal-item').forEach(el => {
      el.addEventListener('dragstart', e => { e.dataTransfer.setData('text/x-akc-type', el.dataset.type); e.dataTransfer.effectAllowed = 'copy'; });
      el.addEventListener('click', () => { if (this.tab === 'sch') this.setTab('board'); this.editor.startPlacing(el.dataset.type); });
    });
  }

  // компоненты из библиотеки EasyEDA, уже добавленные в проект
  renderProjectParts() {
    const box = document.getElementById('palProject');
    if (!box) return;
    const parts = Object.values(this.project.eeParts || {}).filter(p => AK.LIB['ee_' + p.uuid]);
    const key = parts.map(p => p.uuid).join(',');
    if (key === this.palPartsKey) return;
    this.palPartsKey = key;
    box.innerHTML = parts.length ? `<div class="pal-cat">В проекте</div><div class="pal-grid">${parts.map(p => this.palItem(AK.LIB['ee_' + p.uuid])).join('')}</div>` : '';
    this.bindPalItems(box);
  }

  async searchLibrary(q) {
    const out = document.getElementById('eeResults');
    if (!q) { out.innerHTML = ''; return; }
    out.innerHTML = '<div class="pal-help">Поиск…</div>';
    try {
      const list = await AK.EE.search(q);
      const e = AK.util.esc;
      out.innerHTML = list.length ? list.map(r => `<button class="ee-item" data-uuid="${e(r.uuid)}" title="${e(r.title + (r.desc ? ' — ' + r.desc : ''))}">` +
        (r.thumb ? `<img src="${e(r.thumb)}" alt="" loading="lazy">` : '<span class="noimg"></span>') +
        `<span><b>${e(r.title)}</b><small>${e(r.package)}${r.lcsc ? ' · ' + e(r.lcsc) : ''}</small></span></button>`).join('')
        : '<div class="pal-help">Ничего не найдено</div>';
      out.querySelectorAll('.ee-item').forEach(b => b.addEventListener('click', () => this.addLibPart(b.dataset.uuid)));
    } catch (err) { out.innerHTML = `<div class="pal-help">Ошибка поиска: ${AK.util.esc(err.message)}</div>`; }
  }

  async addLibPart(uuid) {
    this.toast('Загрузка компонента из библиотеки EasyEDA…');
    try {
      const part = await AK.EE.fetchPart(uuid);
      this.checkpoint();
      this.project.eeParts = this.project.eeParts || {};
      this.project.eeParts[part.uuid] = part;
      AK.EE.register(this.project);
      const d = AK.LIB['ee_' + part.uuid];
      if (!d) throw new Error('не удалось разобрать символ');
      this.save();
      this.renderProjectParts();
      if (this.tab === 'sch') this.setTab('board');
      this.editor.startPlacing(d.type);
      if (!d.visual.fromFootprint) this.toast('Корпус не для макетки (SMD или шаг не 2.54 мм) — выводы показаны как на переходной плате');
    } catch (err) { this.toast('Не удалось загрузить компонент: ' + err.message); }
  }

  // в EasyEDA (расширение): построить схему прямо в редакторе

    const payload = AK.exportEasyEDA(this.project, m, { native: true });
    window.parent.postMessage({ akc: 'build', payload }, '*');
    this.toast('Создаю схему в EasyEDA…');
  }

  fileBase() { return (this.project.name || 'autokicad').replace(/[\\/:*?"<>|\s]+/g, '_'); }

  exportAs(kind) {
    const base = this.fileBase();
    const nl = this.netlist(), m = this.schematic();
    if (kind !== 'project' && !m.symbols.length) { this.toast('Сначала добавьте компоненты'); return; }
    if (kind === 'kicadzip') {
      const zip = AK.exportKicad.projectZip(this.project, m, base);
      AK.util.download(base + '_kicad.zip', new Blob([zip], { type: 'application/zip' }));
      this.toast('Проект KiCad сохранён (.zip) — распакуйте и откройте .kicad_pro');
    }
    if (kind === 'kicad') { AK.util.download(base + '.kicad_sch', AK.exportKicad.sch(this.project, m)); this.toast('Схема KiCad сохранена (.kicad_sch)'); }
    if (kind === 'net') { AK.util.download(base + '.net', AK.exportKicad.netlist(this.project, nl)); this.toast('Netlist сохранён (.net)'); }
    if (kind === 'easyeda') { AK.util.download(base + '_easyeda.json', AK.exportEasyEDA(this.project, m), 'application/json'); this.toast('Исходник EasyEDA сохранён (.json)'); }
    if (kind === 'svg') { AK.util.download(base + '_schematic.svg', AK.schematicStandaloneSVG(m, this.project.name), 'image/svg+xml'); this.toast('Схема сохранена как SVG'); }
    if (kind === 'project') { AK.util.download(base + '.akc.json', JSON.stringify(this.project, null, 1), 'application/json'); this.toast('Проект сохранён'); }
    if (kind === 'bom') { AK.util.download(base + '_BOM.csv', AK.bomCSV(this.project), 'text/csv;charset=utf-8'); this.toast('Перечень элементов сохранён (.csv)'); }
    if (kind === 'png') {
      AK.util.svgToPng(AK.schematicStandaloneSVG(m, this.project.name), 2)
        .then(blob => { AK.util.download(base + '_schematic.png', blob); this.toast('Схема сохранена как PNG'); })
        .catch(err => this.toast('Не удалось сохранить PNG: ' + err.message));
    }
    if (kind === 'print') this.printSchematic(m);
  }

  // печать схемы; в диалоге печати можно выбрать «Сохранить как PDF»
  printSchematic(m) {
    const e = AK.util.esc, title = e(this.project.name || 'Схема');
    const svg = AK.schematicStandaloneSVG(m, this.project.name).replace(/^<\?xml[^>]*>/, '');
    const f = document.createElement('iframe');
    f.style.cssText = 'position:fixed;right:0;bottom:0;width:0;height:0;border:0';
    document.body.appendChild(f);
    const doc = f.contentDocument;
    doc.open();
    doc.write(`<!doctype html><html><head><meta charset="utf-8"><title>${title}</title>` +
      '<style>@page{size:A4 landscape;margin:10mm}html,body{margin:0}svg{width:100%;height:auto;max-height:180mm}h1{font:600 14px Arial,sans-serif;margin:0 0 6px}</style></head>' +
      `<body><h1>${title}</h1>${svg}</body></html>`);
    doc.close();
    setTimeout(() => {
      try { f.contentWindow.focus(); f.contentWindow.print(); } catch (err) { this.toast('Печать недоступна: ' + err.message); }
      setTimeout(() => f.remove(), 1000);
    }, 200);
  }

  openFile(file) {
    const r = new FileReader();
    r.onload = () => {
      try {
        const p = JSON.parse(r.result);
        if (!this.validate(p)) throw new Error('bad');
        this.checkpoint();
        this.project = p;
        this.select(null);
        this.changed();
        this.editor.fit();
        this.toast('Проект открыт');
      } catch (err) { this.toast('Не удалось открыть файл: это не проект AutoKiCad'); }
    };
    r.readAsText(file);
  }

      if ((e.ctrlKey || e.metaKey) && (k === 'z' || k === 'я')) { e.preventDefault(); e.shiftKey ? this.redo() : this.undo(); return; }
      if ((e.ctrlKey || e.metaKey) && (k === 'y' || k === 'н')) { e.preventDefault(); this.redo(); return; }
      if ((e.ctrlKey || e.metaKey) && (k === 's' || k === 'ы')) { e.preventDefault(); this.exportAs('project'); return; }
      if (this.tab === 'sch') return;
      const ed = this.editor, ctrl = e.ctrlKey || e.metaKey;
      if (ctrl && (k === 'd' || k === 'в')) { e.preventDefault(); ed.duplicateSelection(); return; }
      if (ctrl && (k === 'a' || k === 'ф')) { e.preventDefault(); ed.selectAll(); return; }
      if (ctrl && (k === 'c' || k === 'с')) { if (ed.copySelection()) { e.preventDefault(); this.toast('Скопировано'); } return; }
      if (ctrl && (k === 'x' || k === 'ч')) { if (ed.copySelection()) { e.preventDefault(); ed.deleteSelection(); } return; }
      if (ctrl && (k === 'v' || k === 'м')) { e.preventDefault(); ed.paste(); return; }
      const arrows = { arrowleft: [-1, 0], arrowright: [1, 0], arrowup: [0, -1], arrowdown: [0, 1] };
      if (arrows[k]) { e.preventDefault(); const st = e.shiftKey ? 50 : 10; ed.nudge(arrows[k][0] * st, arrows[k][1] * st); return; }
      if (k === 'escape') { if (!this.editor.cancel()) { this.select(null); this.editor.render(); } return; }
      if (k === 'delete' || k === 'backspace') { e.preventDefault(); this.editor.deleteSelection(); return; }
      if (k === 'r' || k === 'к') { this.editor.rotateSelection(); return; }
      const ci = '12345678'.indexOf(e.key);
      if (ci >= 0) { this.editor.wireColor = ci; this.toast('Цвет нового провода изменён'); }
    });
    window.addEventListener('resize', () => { this.editor.applyView(); if (this.tab !== 'board') this.schView.fit(); });
  }
};

// Новый проект начинается с пустой макетной платы
AK.emptyProject = function () {
  return { version: 1, name: 'Новый проект', components: [{ id: 'bb', type: 'breadboard', ref: 'BB1', value: '', x: 300, y: 250, rot: 0, props: {} }], wires: [] };
};

// Перечень элементов (BOM), как в EasyEDA: одинаковые детали — одной строкой. Разделитель «;» — для Excel.
AK.bomCSV = function (project) {
  const rows = new Map();
  project.components.forEach(c => {
    const d = AK.LIB[c.type];
    if (d.virtual) return;
    const value = c.value || d.value;
    const key = [c.type, value].join('|');
    if (!rows.has(key)) rows.set(key, { refs: [], value, name: d.name, pkg: d.eePackage || '', fp: d.footprint || '', lcsc: d.lcsc || '' });
    rows.get(key).refs.push(c.ref);
  });
  const natural = (a, b) => a.localeCompare(b, undefined, { numeric: true });
  const q = s => '"' + String(s).replace(/"/g, '""') + '"';
  const out = [['№', 'Обозначения', 'Номинал', 'Наименование', 'Корпус', 'Посадочное место KiCad', 'Кол-во', 'LCSC'].map(q).join(';')];
  [...rows.values()].sort((a, b) => natural(a.refs.sort(natural)[0], b.refs.sort(natural)[0]))
    .forEach((r, i) => out.push([i + 1, r.refs.join(', '), r.value, r.name, r.pkg, r.fp, r.refs.length, r.lcsc].map(q).join(';')));
  return '﻿' + out.join('\r\n') + '\r\n';
};
